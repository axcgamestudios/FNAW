   202→  private buildDust() {
   203→    const count = this.quality === 'low' ? 0 : this.quality === 'medium' ? 260 : 420;
   204→    if (count === 0) return;
   205→    const geo = new THREE.BufferGeometry();
   206→    const pos = new Float32Array(count * 3);
   207→    for (let i = 0; i < count; i++) {
   208→      pos[i * 3] = (Math.random() - 0.5) * 16;
   209→      pos[i * 3 + 1] = Math.random() * 3;
   210→      pos[i * 3 + 2] = (Math.random() - 0.5) * 16;
   211→    }
   212→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   213→    const mat = new THREE.PointsMaterial({
   214→      color: 0xbfae8a,
   215→      size: 0.018,
   216→      transparent: true,
   217→      opacity: 0.42,
   218→      depthWrite: false,
   219→      blending: THREE.AdditiveBlending,
   220→    });
   221→    this.dust = new THREE.Points(geo, mat);
   222→    this.scene.add(this.dust);
   223→  }

   225→  private updateDust(_dt: number) {
   226→    if (!this.dust) return;
   227→    this.dust.position.set(
   228→      Math.round(this.player.pos.x / 8) * 8,
   229→      0,
   230→      Math.round(this.player.pos.z / 8) * 8
   231→    );
   232→    const p = this.dust.geometry.getAttribute('position') as THREE.BufferAttribute;
   233→    const arr = p.array as Float32Array;
   234→    for (let i = 0; i < arr.length; i += 3) {
   235→      arr[i + 1] += Math.sin(this.time * 0.6 + i) * 0.0008 + 0.0012;
   236→      arr[i] += Math.cos(this.time * 0.4 + i * 0.7) * 0.0006;
   237→      if (arr[i + 1] > 3.1) arr[i + 1] = 0.05;
   238→    }
   239→    p.needsUpdate = true;
   240→  }

   264→  private menuCamera(dt: number) {
   265→    const t = performance.now() / 1000;
   266→    // cámara fija frente al rostro de Freddy (que está en z=11.2 mirando al sur) con deriva sutil
   267→    this.camera.position.set(Math.sin(t * 0.07) * 0.08, 1.72 + Math.sin(t * 0.11) * 0.02, 12.75);
   268→    this.camera.rotation.set(0, 0, 0);
   269→    this.camera.rotation.order = 'YXZ';
   270→    this.camera.rotation.y = 0 + Math.sin(t * 0.05) * 0.03;
   271→    this.camera.rotation.x = -0.04;
   272→    // espasmos de cabeza de Freddy
   273→    if (this.menuRig) {
   274→      this.menuTwitchT -= dt;
   275→      const head = this.menuRig.bones.head;
   276→      if (head) {
   277→        if (this.menuTwitchT <= 0) {
   278→          this.menuTwitchT = 1.2 + Math.random() * 3.5;
   279→        }
   280→        const twitch = Math.max(0, this.menuTwitchT - (1.2 + Math.random() * 3.5) + 0.4);
   281→        head.rotation.z = Math.sin(t * 0.8) * 0.03 + (Math.random() < 0.02 ? (Math.random() - 0.5) * 0.2 : 0);
   282→        head.rotation.y = Math.sin(t * 0.33) * 0.08;
   283→      }
   284→      for (const e of this.menuRig.eyes) {
   285→        const mats = Array.isArray(e.material) ? e.material : [e.material];
   286→        for (const m of mats as THREE.MeshStandardMaterial[]) {
   287→          m.emissiveIntensity = 1.2 + (Math.random() < 0.06 ? -1.1 : 0);
   288→        }
   289→      }
   290→      this.menuRig.mixer?.update(dt);
   291→    }
   292→    if (this.world) this.world.update(dt, this.camera.position);
   293→  }

   419→  private setupInput() {
   420→    const kd = (e: KeyboardEvent) => this.onKeyDown(e);
   421→    const ku = (e: KeyboardEvent) => this.onKeyUp(e);
   422→    const mm = (e: MouseEvent) => this.onMouseMove(e);
   423→    const rs = () => this.onResize();
   424→    window.addEventListener('keydown', kd as EventListener);
   425→    window.addEventListener('keyup', ku as EventListener);
   426→    window.addEventListener('mousemove', mm as EventListener);
   427→    window.addEventListener('resize', rs);
   428→    this.windowListeners = [
   429→      ['keydown', kd as EventListener], ['keyup', ku as EventListener], ['mousemove', mm as EventListener], ['resize', rs],
   430→    ];

   433→  private onKeyDown(e: KeyboardEvent) {
   434→    if (e.code === 'Tab') e.preventDefault();
   435→    this.player.keys[e.code] = true;
   436→    if (this.mode === 'cinematic') {
   437→      if (e.code === 'KeyE' || e.code === 'Space' || e.code === 'Enter') {
   438→        e.preventDefault();
   439→        this.skipCinematic();
   440→      } else if (e.code === 'Escape') {
   441→        this.cine.stop(true);
   442→      }
   443→      return;
   444→    }
   445→    if (this.mode !== 'playing') {
   446→      if (e.code === 'Escape' && this.mode === 'paused') this.resume();
   447→      return;
   448→    }
   449→    switch (e.code) {
   450→      case 'KeyF':
   451→        if (!this.missions.hasFlashlight) {
   452→          this.notify('La linterna está sobre el escritorio');
   453→          break;
   454→        }
   455→        this.player.toggleFlashlight();
   456→        this.audio.camSwitch();
   457→        break;
   458→      case 'KeyQ':
   459→        this.toggleMask();
   460→        break;
   461→      case 'KeyC':
   462→        this.toggleTablet();
   463→        break;
   464→      case 'ControlLeft':
   465→      case 'ControlRight':
   466→        this.player.crouching = !this.player.crouching;
   467→        break;
   468→      case 'KeyE':
   469→        this.tryInteract();
   470→        break;
   471→      case 'Escape':
   472→        this.pause();
   473→        break;
   474→      case 'ArrowLeft':
   475→      case 'KeyA':
   476→        if (useHud.getState().tabletOpen) this.cams.switchCam(-1);
   477→        break;
   478→      case 'ArrowRight':
   479→      case 'KeyD':
   480→        if (useHud.getState().tabletOpen) this.cams.switchCam(1);
   481→        break;
   482→      default:
   483→        if (e.code.startsWith('Digit')) {
   484→          const n = parseInt(e.code.slice(5), 10);
   485→          const stt = useHud.getState();
   486→          if (stt.tabletOpen && n >= 1 && n <= 9) {
   487→            const cam = this.cams.cams.find((c) => c.index === n);
   488→            if (cam) this.cams.setCam(n);
   489→          }
   490→        }
   491→        break;
   492→    }
   493→  }

   495→  private onKeyUp(e: KeyboardEvent) {
   496→    this.player.keys[e.code] = false;
   497→    if (e.code === 'KeyE') this.holdInteract = null;
   498→  }

   500→  private onMouseMove(e: MouseEvent) {
   501→    if (this.mode !== 'playing') return;
   502→    const locked = document.pointerLockElement === this.canvas;
   503→    if (!locked) {
   504→      if (e.buttons & 1) {
   505→        const settings = useHud.getState().settings;
   506→        this.player.onMouseMove(e.movementX, e.movementY, settings.sensitivity, settings.invertY);
   507→      }
   508→      return;
   509→    }
   510→    const settings = useHud.getState().settings;
   511→    this.player.onMouseMove(e.movementX, e.movementY, settings.sensitivity, settings.invertY);

   773→  private updateInteractions(dt: number) {
   774→    const st = useHud.getState();
   775→    if (this.mode !== 'playing' || st.tabletOpen || st.note || st.keypadDoor) {
   776→      if (st.prompt) st.setPrompt(null);
   777→      this.holdInteract = null;
   778→      return;
   779→    }
   780→    const p = this.player;
   781→    const look = p.lookDir;
   782→    let best: Interactable | null = null;
   783→    let bestScore = -1;
   784→    for (const inter of this.interactables) {
   785→      if (!inter.enabled()) continue;
   786→      const dx = inter.pos.x - p.pos.x;
   787→      const dz = inter.pos.z - p.pos.z;
   788→      const dy = (inter.pos.y ?? 1) - (p.pos.y + 1.5);
   789→      const d = Math.hypot(dx, dz, dy * 0.5);
   790→      if (d > inter.radius) continue;
   791→      let score = 1 - d / inter.radius;
   792→      if (inter.requireLook !== false) {
   793→        const len = Math.hypot(dx, dz) || 1;
   794→        const dot = (dx / len) * look.x + (dz / len) * look.z;
   795→        if (dot < 0.35) continue;
   796→        score += dot * 0.5;
   797→      }
   798→      if (score > bestScore) {
   799→        bestScore = score;
   800→        best = inter;
   801→      }
   802→    }
   803→    // recogidas automáticas (objetos de misión)
   804→    const item = this.missions.itemAt(p.pos, 0.95);
   805→    if (item) {
   806→      this.pickupItem(item.id, item.kind);
   807→    }
   808→    if (best) {
   809→      const holding = this.player.keys['KeyE'];
   810→      if (best.holdTime) {
   811→        if (holding) {
   812→          if (!this.holdInteract || this.holdInteract.inter !== best) {
   813→            this.holdInteract = { inter: best, progress: 0 };
   814→          }
   815→          this.holdInteract.progress += dt / best.holdTime;
   816→          this.addNoise(p.pos.x, p.pos.z, 13);
   817→          if (Math.random() < dt * 6) this.audio.repairClick();
   818→          if (this.holdInteract.progress >= 1) {
   819→            best.action();
   820→            this.holdInteract = null;
   821→            st.setPrompt(null);
   822→          } else {
   823→            st.setPrompt(`${best.holdLabel ?? 'MANTÉN E'} — [E]`, this.holdInteract.progress);
   824→          }
   825→        } else {
   826→          if (this.holdInteract?.inter !== best) this.holdInteract = null;
   827→          st.setPrompt(`${best.label} — mantén [E]`);
   828→        }
   829→      } else {
   830→        this.holdInteract = null;
   831→        st.setPrompt(`${best.label} [E]`);
   832→        if (this.interactPressed) {
   833→          best.action();
   834→          this.interactPressed = false;
   835→        }
   836→      }
   837→    } else {
   838→      if (this.holdInteract) this.holdInteract = null;
   839→      if (st.prompt) st.setPrompt(null);
   840→    }
   841→    if (this.interactPressed) this.interactPressed = false;
   842→  }

   992→  private updateDying(dt: number) {

   998→      ai.pos.addScaledVector(dir, dt * 5.5);

  1099→  private applyQuality() {
  1100→    const dpr = window.devicePixelRatio || 1;
  1101→    const pr = this.quality === 'low' ? 0.7 : this.quality === 'medium' ? Math.min(1, dpr) : Math.min(1.75, dpr);
  1102→    this.renderer?.setPixelRatio(pr);
  1103→    if (this.player) {
  1104→      this.player.flashlight.castShadow = this.quality !== 'low';
  1105→      this.renderer.shadowMap.enabled = this.quality !== 'low';
  1106→    }

  1111→  private render = () => {
  1112→    if (this.disposed) return;
  1113→    this.raf = requestAnimationFrame(this.render);
  1114→    const now = performance.now();
  1115→    let dt = (now - this.lastFrame) / 1000;
  1116→    this.lastFrame = now;
  1117→    if (dt > 0.06) dt = 0.06;
  1118→    if (dt <= 0) return;
  1119→
  1120→    // auto-ajuste de calidad
  1121→    this.fpsAcc += dt;
  1122→    this.fpsCount++;
  1123→    if (this.fpsAcc > 1) {
  1124→      const fps = this.fpsCount / this.fpsAcc;
  1125→      this.fpsAcc = 0;
  1126→      this.fpsCount = 0;
  1127→      if (this.autoQuality && this.mode === 'playing') {
  1128→        if (fps < 32) this.fpsLowTime += 1;
  1129→        else this.fpsLowTime = Math.max(0, this.fpsLowTime - 0.5);
  1130→        if (this.fpsLowTime >= 5) {
  1131→          if (this.quality === 'high') this.quality = 'medium';
  1132→          else if (this.quality === 'medium') this.quality = 'low';
  1133→          this.applyQuality();
  1134→          this.fpsLowTime = 0;
  1135→        }
  1136→      }
  1137→    }
  1138→
  1139→    if (this.mode === 'playing') {
  1140→      this.tick(dt);
  1141→    } else if (this.mode === 'dying') {
  1142→      this.updateDying(dt);
  1143→      this.world.update(dt, this.player.pos);
  1144→    } else if (this.mode === 'cinematic') {
  1145→      this.cine.update(dt);
  1146→      this.world.update(dt, this.camera.position);
  1147→    } else if (this.mode === 'menu' || this.mode === 'boot') {
  1148→      this.menuCamera(dt);
  1149→    } else if (this.mode === 'dead' || this.mode === 'complete' || this.mode === 'ending') {
  1150→      // escena congelada con la última cámara
  1151→      this.world.update(dt, this.camera.position);
  1152→    }
  1153→
  1154→    if (this.audio.available) {
  1155→      this.audio.update(dt);
  1156→      if (this.mode !== 'cinematic') {
  1157→        this.audio.updateListener(this.camera.position, this.camera.quaternion);
  1158→      } else {
  1159→        this.audio.updateListener(this.camera.position, this.camera.quaternion);
  1160→      }
  1161→    }
  1162→
  1163→    const st = useHud.getState();
  1164→    const tabletRendering = st.tabletOpen && this.mode === 'playing';
  1165→    // la tableta renderiza con más exposición (visión nocturna de las cámaras)
  1166→    this.renderer.toneMappingExposure = tabletRendering ? 2.15 : 1.12;
  1167→    const activeCam = tabletRendering ? this.cams.camera : this.camera;
  1168→    this.renderer.render(this.scene, activeCam);
  1169→  };

  1171→  private tick(dt: number) {
  1172→    if (!this.chapterDef || !this.missions) return;
  1173→    const st = useHud.getState();
  1174→    this.time += dt;
  1175→
  1176→    // jugador (bloqueado si tablet/nota/teclado abiertos)
  1177→    const uiBlocked = st.tabletOpen || !!st.note || !!st.keypadDoor;
  1178→    if (!uiBlocked) {
  1179→      this.player.update(dt);
  1180→    } else {
  1181→      this.player.keys = {};
  1182→    }
  1183→
  1184→    // IA host actualizado
  1185→    const host = this.animatronics[0]?.host;
  1186→    if (host) {
  1187→      host.time = this.time;
  1188→      host.chapter = this.chapter;
  1189→      host.chapterTime = this.time;
  1190→      host.aggression = this.chapterDef.aggression;
  1191→    }
  1192→
  1193→    // animatrónicos (solo los del capítulo despiertan; el resto sigue dormido)
  1194→    const playerLike = this.player as unknown as PlayerLike;
  1195→    for (const a of this.animatronics) {
  1196→      if (!a.def.activeChapters.includes(this.chapter) && !a.active) {
  1197→        a.update(dt * 0.2, playerLike); // dormido: animación mínima
  1198→        continue;
  1199→      }
  1200→      a.update(dt, playerLike);
  1201→    }
  1202→
  1203→    // eventos
  1204→    this.events.update(dt);
  1205→    this.events.checkKitchenNoise(this.time / 60);
  1206→    if (!this.events.callDone && this.chapterDef.radio && this.time / 60 >= this.chapterDef.radio.atMinute) {
  1207→      this.events.triggerCall(this.chapterDef.radio.lines);
  1208→    }
  1209→    this.updateCall(dt);
  1210→
  1211→    // objetivos de desplazamiento (los de "volver" solo al final)
  1212→    const room = this.world.roomIdAt(this.player.pos.x, this.player.pos.z);
  1213→    const othersDoneFor = (excludeId: string) =>
  1214→      this.missions.objectives
  1215→        .filter((o) => o.def.id !== excludeId && !o.def.optional && o.def.type !== 'survive')
  1216→        .every((o) => o.done);
  1217→    const gotoChecks: Array<[string, string, boolean]> = [
  1218→      ['c2_kitchen', 'kitchen', false],
  1219→      ['c1_return', 'office', true],
  1220→      ['c2_return', 'office', true],
  1221→      ['c3_survive', 'office', false],
  1222→    ];
  1223→    for (const [objId, roomId, requireOthers] of gotoChecks) {
  1224→      const o = this.missions.objectives.find((x) => x.def.id === objId);
  1225→      if (o && !o.done && room === roomId && (!requireOthers || othersDoneFor(o.def.id))) {
  1226→        this.missions.completeObjective(objId);
  1227→      }
  1228→    }
  1229→
  1230→    // energía
  1231→    let drain = 0.045;
  1232→    if (st.tabletOpen) drain += 0.29;
  1233→    if (this.player.flashlightOn) drain += 0.05;
  1234→    if (this.player.running) drain += 0.06;
  1235→    this.power = Math.max(0, this.power - drain * dt);
  1236→    this.world.setPowerFactor(this.power > 25 ? 1 : this.power / 25);
  1237→    if (this.power <= 0) {
  1238→      for (const r of this.world.rooms.values()) r.lightOn = false;
  1239→    } else if (this.power > 3) {
  1240→      for (const r of this.world.rooms.values()) r.lightOn = true;
  1241→    }
  1242→
  1243→    // puertas animadas
  1244→    for (const d of this.world.doors.values()) {
  1245→      if (d.angle !== d.targetAngle) {
  1246→        const diff = d.targetAngle - d.group.rotation.y;
  1247→        d.group.rotation.y += Math.max(-3.2 * dt, Math.min(3.2 * dt, diff));
  1248→        d.angle = d.group.rotation.y;
  1249→        d.collider.active = Math.abs(d.angle - (d.def.axis === 'x' ? 0 : Math.PI / 2)) < 0.5 && !d.open;
  1250→        if (Math.abs(diff) < 0.02) {
  1251→          d.angle = d.targetAngle;
  1252→          d.group.rotation.y = d.targetAngle;
  1253→        }
  1254→      }
  1255→    }
  1256→
  1257→    // tapas de conducto: Bonnie las tumba
  1258→    for (const v of this.world.vents.values()) {
  1259→      if (!v.coverClosed) continue;
  1260→      const bonnie = this.animatronics.find((a) => a.def.id === 'bonnie');
  1261→      if (bonnie && bonnie.active) {
  1262→        const px = v.def.coverAt === 'a' ? v.def.a[0] : v.def.b[0];
  1263→        const pz = v.def.coverAt === 'a' ? v.def.a[1] : v.def.b[1];
  1264→        const d = Math.hypot(bonnie.pos.x - px, bonnie.pos.z - pz);
  1265→        if (d < 1.6 && (bonnie.state === 'chase' || bonnie.state === 'hunt')) {
  1266→          v.coverTimer -= dt;
  1267→          if (v.coverTimer < 0 && Math.random() < dt * 1.4) {
  1268→            this.audio.ventBang({ x: px, y: 0.6, z: pz });
  1269→            v.coverTimer = 2.4;
  1270→          }
  1271→          if (Math.random() < dt * 0.11) {
  1272→            v.coverClosed = false;
  1273→            v.blocked = false;
  1274→            if (v.coverMesh) v.coverMesh.visible = false;
  1275→            if (v.coverCollider) v.coverCollider.active = false;
  1276→            this.nav.setVentBlocked(v.def.id, false);
  1277→            this.audio.doorSlam({ x: px, y: 0.6, z: pz });
  1278→            this.notify('¡TAPA DE CONDUCTO DERRIBADA!');
  1279→          }
  1280→        }
  1281→      }
  1282→    }
  1283→
  1284→    // detección de escondites
  1285→    if (this.player.hidden) {
  1286→      for (const a of this.animatronics) {
  1287→        if (!a.active || a.state === 'dormant') continue;
  1288→        const d = a.pos.distanceTo(this.player.pos);
  1289→        if (d < 2.0 && (a.state === 'search' || a.state === 'chase' || a.state === 'hunt')) {
  1290→          if (Math.random() < dt * 0.35) {
  1291→            const spot = mapData.hidingSpots.find((s) => s.id === this.player.hiddenSpotId);
  1292→            this.player.unhide(new THREE.Vector3(spot ? spot.pos[0] : this.player.pos.x, 0, spot ? spot.pos[1] : this.player.pos.z));
  1293→            a.state = 'chase';
  1294→            a.stateTimer = 0;
  1295→            this.notify('¡TE HAN ENCONTRADO!');
  1296→            break;
  1297→          }
  1298→        }
  1299→      }
  1300→    }
  1301→
  1302→    // peligro
  1303→    let danger = 0;
  1304→    let chase = false;
  1305→    for (const a of this.animatronics) {
  1306→      if (!a.active) continue;
  1307→      if (a.state === 'chase' || a.state === 'hunt' || a.state === 'attack') {
  1308→        chase = true;
  1309→        danger = Math.max(danger, 1);
  1310→      } else {
  1311→        const d = a.pos.distanceTo(this.player.pos);
  1312→        danger = Math.max(danger, Math.max(0, 1 - d / 14) * 0.5);
  1313→      }
  1314→    }
  1315→    this.danger = danger;
  1316→    this.audio.danger = danger;
  1317→    this.audio.setChase(chase);
  1318→
  1319→    // mundo (luces)
  1320→    const focus = st.tabletOpen ? this.cams.camera.position : this.player.pos;
  1321→    this.world.update(dt, focus);
  1322→    this.updateDust(dt);
  1323→    this.updateInteractions(dt);
  1324→    this.cams.update(dt, this.animatronics, st.tabletOpen);
  1325→
  1326→    // fin de capítulo: todos los objetivos requeridos
  1327→    if (!this.chapterEndPending && this.missions.allRequiredDone()) {
  1328→      this.completeChapter();
  1329→      return;
  1330→    }
  1331→
  1332→    // HUD a 10 Hz
  1333→    this.hudTimer -= dt;
  1334→    if (this.hudTimer <= 0) {
  1335→      this.hudTimer = 0.1;
  1336→      useHud.getState().syncHud({
  1337→        battery: this.player.flashlightBattery,
  1338→        stamina: this.player.stamina,
  1339→        power: this.power,
  1340→        maskOn: this.player.masked,
  1341→        flashlightOn: this.player.flashlightOn,
  1342→        crouching: this.player.crouching,
  1343→        inVent: this.player.inVent,
  1344→        danger,
  1345→        tabletGlitch: this.cams.glitchLevel > 0.55,
  1346→      });
  1347→    }
  1348→  }
  1349→
  1350→  /* ============ LIMPIEZA ============ */
  1351→
  1352→  dispose() {
  1353→    this.disposed = true;
  1354→    cancelAnimationFrame(this.raf);
  1355→    for (const [name, fn] of this.windowListeners) {
  1356→      window.removeEventListener(name, fn);
  1357→    }
  1358→    if (document.pointerLockElement === this.canvas) document.exitPointerLock();
  1359→    this.audio.suspend();
  1360→    this.scene.traverse((o) => {
  1361→      if (o instanceof THREE.Mesh) {
  1362→        o.geometry.dispose();
  1363→        const mats = Array.isArray(o.material) ? o.material : [o.material];
  1364→        for (const m of mats) m.dispose();
  1365→      }
  1366→    });
  1367→    disposeTextures();
  1368→    disposeMaterials();
  1369→    this.renderer?.dispose();
  1370→  }
  1371→}
