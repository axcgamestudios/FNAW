    85→/** ajusta el brillo de los halos de ojos de un rig */
    86→export function setEyeGlow(rig: { eyeGlow: THREE.Mesh | null; eyeGlows?: THREE.Mesh[] }, level: number) {
    87→  const list = rig.eyeGlows && rig.eyeGlows.length ? rig.eyeGlows : rig.eyeGlow ? [rig.eyeGlow] : [];
    88→  for (const e of list) {
    89→    const m = e.material as THREE.MeshBasicMaterial;
    90→    m.opacity = Math.min(1, 0.25 + level * 0.75);
    91→    const base = (e.userData as { baseScale?: number }).baseScale ?? 1;
    92→    e.scale.setScalar(base * (1 + level * 0.35));
    93→  }
    94→}

   152→  /** Construye el rig: GLTF del repo si está disponible; procedural si no */
   153→  private buildRig(def: AnimatronicDef): ARig {
   154→    const assetKind = def.kind; // 'freddy' | 'bonnie' | 'chica' | 'foxy' | 'golden' | 'luna'
   155→    let gltfRig: GltfRig | null = null;
   156→    if (assetKind === 'golden') {
   157→      gltfRig = Assets.createVariant('golden', 0xc8a028);
   158→    } else if (assetKind === 'luna') {
   159→      // LUNA: modelo GLB estático del Eclipse (moon.glb) — vuela/rueda, no camina
   160→      gltfRig = Assets.rigs.get('moon') ?? null;
   161→    } else if (Assets.has(assetKind)) {
   162→      gltfRig = Assets.rigs.get(assetKind) ?? null;
   163→    }
   164→    if (gltfRig) {
   165→      /* v11 — HALOS DE OJOS CORRECTOS:
   166→       * 1. Se muestrean los VÉRTICES SKINNED del mesh de ojos (applyBoneTransform),
   167→       *    no la caja del nodo (que en Freddy cae bajo el suelo y en Bonnie se
   168→       *    quedaba flotando quieta en la "bola morada").
   169→       * 2. Si el mesh contiene ambos ojos, se separa en 2 grupos por X → un glow por ojo.
   170→       * 3. Cada glow se ANCLA AL HUESO más cercano que gobierna esa zona
   171→       *    (huesos de ojo si existen, si no el hueso cabeza) → sigue la
   172→       *    animación (cabeceo, giro, persecución) en vez de quedarse flotando. */
   173→      const eyeGlows: THREE.Mesh[] = [];
   174→      const glowMat = new THREE.MeshBasicMaterial({
   175→        color: def.eyeColor,
   176→        transparent: true,
   177→        opacity: 0.55,
   178→        blending: THREE.AdditiveBlending,
   179→        depthWrite: false,
   180→        toneMapped: false,
   181→      });
   182→      gltfRig.root.updateMatrixWorld(true);
   183→
   184→      // huesos candidatos para anclar: ojos por nombre + cabeza
   185→      const eyeBones: THREE.Object3D[] = [];
   186→      gltfRig.model.traverse((o) => {
   187→        if (!(o as THREE.Mesh).isMesh && /eye/i.test(o.name)) eyeBones.push(o);
   188→      });
   189→      const anchorCandidates: THREE.Object3D[] = [...eyeBones];
   190→      if (gltfRig.bones.head) anchorCandidates.push(gltfRig.bones.head);
   191→
   192→      // muestrear centros de ojos a partir de vértices skinned (en mundo)
   193→      const sampleEyeCenters = (mesh: THREE.Mesh): { centers: THREE.Vector3[]; radius: number } => {
   194→        const pos = mesh.geometry.attributes.position as THREE.BufferAttribute | undefined;
   195→        if (!pos) return { centers: [], radius: 0.03 };
   196→        const v = new THREE.Vector3();
   197→        const pts: THREE.Vector3[] = [];
   198→        const step = Math.max(1, Math.floor(pos.count / 220));
   199→        for (let i = 0; i < pos.count; i += step) {
   200→          v.fromBufferAttribute(pos, i);
   201→          if ((mesh as THREE.SkinnedMesh).isSkinnedMesh) {
   202→            try { (mesh as THREE.SkinnedMesh).applyBoneTransform(i, v); } catch { /* noop */ }
   203→          }
   204→          mesh.localToWorld(v);
   205→          pts.push(v.clone());
   206→        }
   207→        if (!pts.length) return { centers: [], radius: 0.03 };
   208→        // caja real de los vértices muestreados
   209→        const bb = new THREE.Box3().setFromPoints(pts);
   210→        const size = bb.getSize(new THREE.Vector3());
   211→        const radius = Math.max(0.024, Math.min(size.y, size.z) * 0.36);
   212→        const split = size.x > size.y * 1.55; // ambos ojos en un solo mesh
   213→        const centers: THREE.Vector3[] = [];
   214→        if (split) {
   215→          const midX = (bb.min.x + bb.max.x) / 2;
   216→          for (const side of [-1, 1]) {
   217→            const half = pts.filter((p) => (side < 0 ? p.x < midX : p.x >= midX));
   218→            if (half.length) centers.push(new THREE.Box3().setFromPoints(half).getCenter(new THREE.Vector3()));
   219→          }
   220→        } else {
   221→          centers.push(bb.getCenter(new THREE.Vector3()));
   222→        }
   223→        return { centers, radius };
   224→      };
   225→
   226→      for (const eye of gltfRig.eyes) {
   227→        const { centers, radius } = sampleEyeCenters(eye);
   228→        for (const cWorld of centers) {
   229→          // hueso de anclaje: el candidato más cercano al centro del ojo
   230→          let anchor: THREE.Object3D = eye;
   231→          let bestD = Infinity;
   232→          for (const cand of anchorCandidates) {
   233→            const cw = cand.getWorldPosition(new THREE.Vector3());
   234→            const d = cw.distanceTo(cWorld);
   235→            if (d < bestD && d < 0.9) { bestD = d; anchor = cand; }
   236→          }
   237→          const glow = new THREE.Mesh(new THREE.SphereGeometry(radius, 10, 8), glowMat);
   238→          const inv = 1 / Math.max(0.0001, anchor.getWorldScale(new THREE.Vector3()).x);
   239→          glow.position.copy(anchor.worldToLocal(cWorld.clone()));
   240→          glow.scale.setScalar(inv);
   241→          (glow.userData as { baseScale?: number }).baseScale = inv;
   242→          anchor.add(glow);
   243→          eyeGlows.push(glow);
   244→        }
   245→      }
   246→      if (!eyeGlows.length && def.kind === 'luna') {
   247→        // Luna (GLB sin meshes de ojo): brillo tenue de la cara, pegado al frente
   248→        const glow = new THREE.Mesh(new THREE.SphereGeometry(0.075, 10, 8), glowMat);
   249→        glow.position.set(0, gltfRig.height * 0.78, 0.12);
   250→        gltfRig.root.add(glow);
   251→        eyeGlows.push(glow);
   252→      }
   253→      const rig: ARig = {
   254→        root: gltfRig.root,
   255→        model: gltfRig.model,
   256→        bones: gltfRig.bones,
   257→        eyes: gltfRig.eyes,
   258→        eyeGlows,
   259→        eyeGlow: eyeGlows[0] ?? null,
   260→        height: gltfRig.height,
   261→      };
   262→      this.moonRig = def.kind === 'luna' ? gltfRig : null;
   263→      return rig;
   264→    }
   265→    // fallback procedural
   266→    const char = buildAnimatronic(
   267→      def.kind === 'freddy' || def.kind === 'golden' ? 'bear'
   268→        : def.kind === 'bonnie' ? 'rabbit'
   269→        : def.kind === 'chica' ? 'chicken'
   270→        : def.kind === 'luna' ? 'solluna' : 'fox',
   271→      def.eyeColor
   272→    );
   273→    const rig: ARig = {
   274→      root: char.root,
   275→      model: null,
   276→      bones: char.bones,
   277→      eyes: char.eyes,
   278→      eyeGlows: char.eyeGlow ? [char.eyeGlow] : [],
   279→      eyeGlow: char.eyeGlow ?? null,
   280→      height: char.height,
   281→    };
   282→    this.charRig = char;
   283→    return rig;
   284→  }

   416→  /* ================== ACTUALIZACIÓN PRINCIPAL ================== */
   417→
   418→  update(dt: number, player: PlayerLike) {
   419→    this.stateTimer += dt;
   420→    this.thinkTimer -= dt;
   421→    this.vocalTimer -= dt;
   422→
   423→    if (!this.active) {
   424→      // retirado: ni caza ni despierta (Sol permanente)
   425→      if (this.retired) {
   426→        this.currentSpeed = 0;
   427→        this.updateAnimator(dt, player, 'dormant');
   428→        return;
   429→      }
   430→      // despertar por temporizador del capítulo
   431→      const delay = this.wakeDelayForChapter();
   432→      if (delay < 900 && this.host.chapterTime >= delay) {
   433→        this.wake();
   434→      } else {
   435→        this.currentSpeed = 0;
   436→        this.updateAnimator(dt, player, 'dormant');
   437→        return;
   438→      }
   439→    }
   440→
   441→    // vocalizaciones aleatorias
   442→    if (this.vocalTimer <= 0) {
   443→      this.vocalTimer = 14 + Math.random() * 22;
   444→      if (this.pos.distanceTo(player.pos) < 18) {
   445→        this.host.sfx.vocal(this.pos, this.def.voice);
   446→      }
   447→    }
   448→
   449→    // detección
   450→    if (this.thinkTimer <= 0) {
   451→      this.thinkTimer = 0.12 + Math.random() * 0.08;
   452→      this.think(player);
   453→    }
   454→
   455→    // linterna
   456→    this.flashlightCheck(dt, player);
   457→
   458→    // máquina de estados
   459→    switch (this.state) {
   460→      case 'patrol': this.updatePatrol(dt, player); break;
   461→      case 'investigate': this.updateInvestigate(dt, player); break;
   462→      case 'chase': this.updateChase(dt, player); break;
   463→      case 'hunt': this.updateHunt(dt, player); break;
   464→      case 'search': this.updateSearch(dt, player); break;
   465→      case 'checkMask': this.updateCheckMask(dt, player); break;
   466→      case 'stunned': this.updateStunned(dt); break;
   467→      case 'attack': this.updateAttack(dt, player); break;
   468→      case 'retreat': this.updateRetreat(dt, player); break;
   469→      default: this.currentSpeed = Math.max(0, this.currentSpeed - dt * 4); break;
   470→    }
   471→
   472→    // física mínima: y del suelo
   473→    const fy = this.host.world.floorHeightAt(this.pos.x, this.pos.z);
   474→    this.pos.y += (fy - this.pos.y) * Math.min(1, dt * 10);
   475→    this.rig.root.position.copy(this.pos);
   476→    if (this.hanging) this.rig.root.position.y = this.hangY;
   477→    if (!this.crawling) this.rig.root.rotation.set(0, this.yaw, 0);
   478→    else this.rig.root.rotation.set(0.55, this.yaw, 0.12);
   479→
   480→    // LUNA (GLB del Eclipse): no camina — PLANEA. Flotación del RigAnimator
   481→    // + giro de muñeco y vuelco completo cuando cuelga del techo.
   482→    if (this.def.kind === 'luna' && this.rig.model) {
   483→      const m = this.rig.model;
   484→      const t = this.host.time;
   485→      const speedT = Math.min(1, this.currentSpeed / Math.max(0.1, this.def.speedWalk));
   486→      // vuelco: colgada boca abajo (π) ↔ de pie (0, pose del GLB ya erguido)
   487→      const wantX = this.hanging ? Math.PI : 0;
   488→      m.rotation.x += (wantX - m.rotation.x) * Math.min(1, dt * 2.2);
   489→      // giro de muñeco (sacudida espasmódica al cazar)
   490→      m.rotation.z = this.hanging
   491→        ? Math.sin(t * 0.7) * 0.55
   492→        : Math.sin(t * 0.85) * 0.2 + (this.state === 'chase' ? Math.sin(t * 14) * 0.06 : 0);
   493→      // se inclina hacia delante al avanzar (planeo)
   494→      if (!this.crawling) this.rig.root.rotation.x = this.hanging ? 0 : speedT * 0.17;
   495→    }
   496→
   497→    this.updateAnimator(dt, player);
   498→  }

   513→  private think(player: PlayerLike) {
   514→    const d = this.pos.distanceTo(player.pos);
   515→    // ruido
   516→    if (this.state === 'patrol' || this.state === 'search' || this.state === 'retreat' || this.state === 'investigate') {
   517→      for (const n of this.host.noises) {
   518→        const age = this.host.time - n.worldTime;
   519→        if (age > 0 && age < 1.2) {
   520→          const dn = Math.hypot(this.pos.x - n.x, this.pos.z - n.z);
   521→          const radius = n.radius * this.def.hearing * (0.8 + this.host.aggression * 0.3);
   522→          if (dn < radius) {
   523→            this.investigatePoint = new THREE.Vector3(n.x, 0, n.z);
   524→            if (this.state !== 'investigate') {
   525→              this.state = 'investigate';
   526→              this.stateTimer = 0;
   527→              this.host.sfx.vocal(this.pos, this.def.voice);
   528→            }
   529→            return;
   530→          }
   531→        }
   532→      }
   533→    }
   534→
   535→    // visión
   536→    if (this.canSee(player)) {
   537→      this.lastKnown = player.pos.clone();
   538→      this.lastSeenAt = this.host.time;
   539→      if (this.state === 'hunt' || this.state === 'chase' || this.state === 'attack' || this.state === 'checkMask') {
   540→        // ya está sobre el jugador
   541→      } else {
   542→        const dist = this.pos.distanceTo(player.pos);
   543→        if (player.masked && this.def.maskFoolable && dist < 4.2 && dist > 1.2 && !this.maskLocked) {
   544→          this.state = 'checkMask';
   545→          this.stateTimer = 0;
   546→          this.fooledTime = 0;
   547→          this.host.sfx.servo(this.pos, this.def.voice);
   548→          return;
   549→        }
   550→        if (player.masked && !this.def.maskFoolable) {
   551→          this.host.sfx.roar(this.pos, this.def.voice);
   552→        }
   553→        this.startChase();
   554→      }
   555→    } else if (this.state === 'chase' && this.host.time - this.lastSeenAt > 2.6) {
   556→      this.state = 'search';
   557→      this.stateTimer = 0;
   558→    }
   559→  }
   560→
   561→  canSee(player: PlayerLike): boolean {
   562→    if (player.hidden) return false;
   563→    const d = this.pos.distanceTo(player.pos);
   564→    let maxDist = this.def.visionDist * (0.75 + this.host.aggression * 0.35);
   565→    const light = player.roomLightLevel();
   566→    if (!player.flashlightOn && light < 0.25) maxDist *= 0.45;
   567→    if (player.crouching) maxDist *= 0.8;
   568→    if (player.flashlightOn) maxDist *= 1.25;
   569→    if (d > maxDist) return false;
   570→    if (this.def.visionFov < 360) {
   571→      _v1.subVectors(player.pos, this.pos);
   572→      _v1.y = 0;
   573→      _v1.normalize();
   574→      const facing = _v2.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   575→      const dot = _v1.dot(facing);
   576→      const cosHalf = Math.cos((this.def.visionFov * Math.PI) / 360);
   577→      if (dot < cosHalf && d > 2.2) return false;
   578→    }
   579→    return this.host.nav.los(this.pos, player.pos, this.def.ventAccess);
   580→  }

   584→  private flashlightCheck(dt: number, player: PlayerLike) {
   585→    if (!player.flashlightOn) {
   586→      this.litTime = Math.max(0, this.litTime - dt);
   587→      return;
   588→    }
   589→    if (this.def.flashlight === 'none' || this.state === 'attack' || this.state === 'hunt') return;
   590→    const d = this.pos.distanceTo(player.pos);
   591→    if (d > 13) return;
   592→    _v1.subVectors(this.pos, player.pos).normalize();
   593→    if (_v1.dot(player.lookDir) < 0.955) return;
   594→    if (!this.host.nav.los(player.pos, this.pos, this.def.ventAccess)) return;
   595→    this.litTime += dt;
   596→    if (this.def.flashlight === 'freeze' && this.litTime > 0.1) {
   597→      if (this.state !== 'stunned' && this.state !== 'checkMask') {
   598→        this.state = 'stunned';
   599→        this.stateTimer = 0;
   600→        this.host.sfx.servo(this.pos, this.def.voice);
   601→      }
   602→    }
   603→    if (this.litTime > 2.4) {
   604→      this.litTime = 0;
   605→      this.retreat();
   606→      this.host.sfx.vocal(this.pos, this.def.voice);
   607→    }
   608→  }

   624→  private updatePatrol(dt: number, player: PlayerLike) {
   625→    if (!this.patrolTarget || this.arrived(1.2) || this.stateTimer > 40) {
   626→      const rooms = this.def.patrolRooms;
   627→      const room = rooms[(Math.random() * rooms.length) | 0];
   628→      const r = this.host.world.rooms.get(room);
   629→      if (r) {
   630→        const [x1, z1, x2, z2] = r.def.rect;
   631→        let tx = x1 + 1 + Math.random() * (x2 - x1 - 2);
   632→        let tz = z1 + 1 + Math.random() * (z2 - z1 - 2);
   633→        if (Math.random() < 0.28 * this.host.aggression) {
   634→          const pr = this.host.world.roomAt(player.pos.x, player.pos.z);
   635→          if (pr && (rooms.includes(pr.def.id) || Math.random() < 0.3)) {
   636→            tx = player.pos.x + (Math.random() - 0.5) * 3;
   637→            tz = player.pos.z + (Math.random() - 0.5) * 3;
   638→          }
   639→        }
   640→        this.patrolTarget = new THREE.Vector3(tx, 0, tz);
   641→        this.setPath(this.patrolTarget);
   642→        this.stateTimer = 0;
   643→      }
   644→    }
   645→    this.followPath(dt, this.def.speedWalk * (0.85 + this.host.aggression * 0.25), player);
   646→  }

   648→  private updateInvestigate(dt: number, player: PlayerLike) {
   649→    if (this.investigatePoint) {
   650→      if (!this.path.length || this.arrived(1.4) || this.thinkTimer > 0.5) {
   651→        if (this.pos.distanceTo(this.investigatePoint) > 1.6 && this.stateTimer < 20) {
   652→          this.setPath(this.investigatePoint);
   653→        } else {
   654→          this.yaw += Math.sin(this.stateTimer * 2.2) * dt * 2.5;
   655→          if (this.stateTimer > 6) {
   656→            this.investigatePoint = null;
   657→            this.state = 'patrol';
   658→            this.stateTimer = 0;
   659→            this.patrolTarget = null;
   660→          }
   661→        }
   662→      }
   663→    } else {
   664→      this.state = 'patrol';
   665→    }
   666→    if (this.path.length) {
   667→      this.followPath(dt, this.def.speedWalk * 1.15, player);
   668→    } else {
   669→      this.currentSpeed = Math.max(0, this.currentSpeed - dt * 4);
   670→    }
   671→  }

   677→  private updateChase(dt: number, player: PlayerLike) {
   678→    const repath = this.stateTimer > 1.6 || !this.path.length;
   679→    if (repath) {
   680→      this.stateTimer = 0.01;
   681→      if (this.canSee(player) || this.def.persistent) {
   682→        this.lastKnown = player.pos.clone();
   683→        this.lastSeenAt = this.host.time;
   684→      }
   685→      this.repathToPlayer();
   686→    }
   687→    const speed = this.def.speedRun * (0.9 + this.host.aggression * 0.18);
   688→    this.followPath(dt, speed, player);
   689→    const d = this.pos.distanceTo(player.pos);
   690→    if (d < 2.8 && d > 1.1 && player.masked && this.def.maskFoolable && !this.maskLocked) {
   691→      if (Math.random() < 0.55) {
   692→        this.state = 'checkMask';
   693→        this.stateTimer = 0;
   694→        this.fooledTime = 0;
   695→        this.host.sfx.servo(this.pos, this.def.voice);
   696→        return;
   697→      }
   698→      this.maskLocked = true;
   699→    }
   700→    if (d <= this.def.attackRange && !player.hidden) {
   701→      this.beginAttack(player);
   702→    }
   703→    if (this.stateTimer > 26) {
   704→      this.state = 'search';
   705→      this.stateTimer = 0;
   706→    }
   707→  }

   709→  private updateHunt(dt: number, player: PlayerLike) {
   710→    this.lastKnown = player.pos.clone();

   712→      this.stateTimer = 0.01;
   713→      this.setPath(this.lastKnown);
   714→    }
   715→    const speed = this.def.speedRun * (0.85 + this.host.aggression * 0.2);
   716→    this.followPath(dt, speed, player);
   717→    const d = this.pos.distanceTo(player.pos);
   718→    if (d <= this.def.attackRange && !player.hidden) {
   719→      this.beginAttack(player);
   720→    }

   723→  private updateSearch(dt: number, player: PlayerLike) {
   724→    if (!this.lastKnown) {

   726→      return;
   727→    }
   728→    if (!this.path.length || this.arrived(1.5)) {
   729→      const angle = Math.random() * Math.PI * 2;
   730→      const r = 2 + Math.random() * 5;
   731→      const target = new THREE.Vector3(
   732→        this.lastKnown.x + Math.cos(angle) * r,
   733→        0,
   734→        this.lastKnown.z + Math.sin(angle) * r
   735→      );

   737→      if (this.stateTimer > 14) {
   738→        this.state = 'patrol';
   739→        this.stateTimer = 0;
   740→        this.patrolTarget = null;
   741→      }
   742→    }
   743→    this.followPath(dt, this.def.speedWalk * 1.3, player);
   744→  }

   773→    _v1.subVectors(player.pos, this.pos);
   774→    _v1.y = 0;

   779→  private updateAttack(dt: number, player: PlayerLike) {
   780→    const windup = this.def.attackWindup;
   781→    if (this.stateTimer < windup) {
   782→      _v1.subVectors(player.pos, this.pos);
   783→      this.yaw = Math.atan2(_v1.x, _v1.z);
   784→      const d = this.pos.distanceTo(player.pos);
   785→      if (d > this.def.attackRange + 1.1) {
   786→        this.startChase();
   787→        return;
   788→      }
   789→      return;
   790→    }
   791→    const speed = 9;
   792→    this.pos.addScaledVector(this.lungeDir, speed * dt);
   793→    this.rig.root.position.copy(this.pos);
   794→    const d = this.pos.distanceTo(player.pos);
   795→    if (d < 0.85 || this.stateTimer > windup + 0.42) {
   796→      this.host.onDeath(this);
   797→    }
   798→  }

   814→      this.state = 'patrol';
   815→      this.stateTimer = 0;
   816→      this.patrolTarget = null;
   817→    } else {
   818→      this.followPath(dt, this.def.speedRun * 0.85, player);
   819→    }

   824→  private setPath(target: THREE.Vector3) {
   825→    this.path = this.host.nav.findPath(this.pos, target, this.def.ventAccess);
   826→    this.pathIdx = 0;
   827→    this.stuckTimer = 0;
   828→    if (!this.path.length) {
   829→      const rooms = this.def.patrolRooms;
   830→      for (const roomId of rooms) {
   831→        const r = this.host.world.rooms.get(roomId);
   832→        if (r) {
   833→          const alt = new THREE.Vector3(r.center.x, 0, r.center.z);
   834→          const p = this.host.nav.findPath(this.pos, alt, this.def.ventAccess);
   835→          if (p.length) {
   836→            this.path = p;
   837→            this.pathIdx = 0;
   838→            return;
   839→          }
   840→        }
   841→      }
   842→    }
   843→  }

   845→  private arrived(r: number): boolean {
   846→    if (!this.path.length) return true;
   847→    const last = this.path[this.path.length - 1];
   848→    return Math.hypot(this.pos.x - last.x, this.pos.z - last.z) < r;
   849→  }

   851→  private followPath(dt: number, speed: number, player: PlayerLike) {
   852→    if (!this.path.length || this.pathIdx >= this.path.length) {
   853→      this.currentSpeed = Math.max(0, this.currentSpeed - dt * 6);
   854→      return;
   855→    }
   856→    const target = this.path[this.pathIdx];
   857→    _v1.set(target.x - this.pos.x, 0, target.z - this.pos.z);
   858→    const d = _v1.length();
   859→    if (d < 0.35) {
   860→      this.pathIdx++;
   861→      return;
   862→    }
   863→    _v1.normalize();
   864→    const targetYaw = Math.atan2(_v1.x, _v1.z);
   865→    let diff = targetYaw - this.yaw;
   866→    while (diff > Math.PI) diff -= Math.PI * 2;
   867→    while (diff < -Math.PI) diff += Math.PI * 2;
   868→    this.yaw += Math.max(-4.5 * dt, Math.min(4.5 * dt, diff));
   869→
   870→    const cellVent = this.isVentCell(this.pos.x, this.pos.z);
   871→    const targetVent = this.isVentCell(target.x, target.z);
   872→    const wantCrawl = cellVent || targetVent;
   873→    const crawlSpeedMul = wantCrawl ? 0.55 : 1;
   874→    if (this.crawling !== wantCrawl) this.crawling = wantCrawl;
   875→
   876→    const v = speed * crawlSpeedMul;
   877→    this.currentSpeed = v;
   878→    const step = v * dt;
   879→    this.pos.x += _v1.x * step;
   880→    this.pos.z += _v1.z * step;
   881→
   882→    if (step < v * dt * 0.35) {
   883→      this.stuckTimer += dt;
   884→    } else {
   885→      this.stuckTimer = Math.max(0, this.stuckTimer - dt);
   886→    }
   887→    if (this.stuckTimer > 0.8 && this.crawling) {
   888→      this.host.sfx.ventBang(this.pos);
   889→      this.stuckTimer = 0;
   890→    }
   891→
   892→    this.autoDoors();
   893→
   894→    this.footTimer -= dt * (v > this.def.speedWalk * 1.05 ? 1.6 : 1);
   895→    if (this.footTimer <= 0) {
   896→      this.footTimer = 0.55 + Math.random() * 0.1;
   897→      const surface = this.crawling ? 'metal' : this.surfaceAt();
   898→      this.host.sfx.footstep(this.pos, surface, v > this.def.speedWalk * 1.05, true);
   899→    }
   900→  }
   901→
   902→  private isVentCell(x: number, z: number): boolean {
   903→    const nav = this.host.nav;
   904→    const i = nav.cellX(x);
   905→    const j = nav.cellZ(z);
   906→    if (i < 0 || j < 0 || i >= GRID_W || j >= GRID_H) return false;
   907→    return nav.grid[j * GRID_W + i] === 2;
   908→  }
   909→
   910→  private surfaceAt(): string {
   911→    const room = this.host.world.roomAt(this.pos.x, this.pos.z);
   912→    if (!room) return 'tile';
   913→    switch (room.def.floor) {
   914→      case 'wood': return 'wood';
   915→      case 'metal': return 'metal';
   916→      case 'carpet': return 'carpet';
   917→      case 'concrete': return 'concrete';
   918→      default: return 'tile';
   919→    }
   920→  }
   921→
   922→  private autoDoors() {
   923→    for (const d of this.host.world.doors.values()) {
   924→      if (d.open || d.locked) continue;
   925→      const dd = Math.hypot(d.def.x - this.pos.x, d.def.z - this.pos.z);
   926→      if (dd < 1.6) {
   927→        d.open = true;
   928→        d.targetAngle = d.group.rotation.y + (Math.random() > 0.5 ? 1.9 : -1.9);
   929→        d.opening = true;
   930→        this.host.sfx.doorCreak(this.pos);
   931→      }
   932→    }
   933→  }

   935→  /* ================== ANIMACIÓN (RigAnimator en espacio de personaje) ================== */

   937→  private updateAnimator(dt: number, player: PlayerLike | null, stateOverride?: string) {
   938→    const t = this.host.time;
   939→    let lookYaw = 0;
   940→    let lookPitch = 0;
   941→    if (player) {
   942→      const d = this.pos.distanceTo(player.pos);
   943→      const interested =
   944→        stateOverride === 'dying' || this.state === 'chase' || this.state === 'hunt' ||
   945→        this.state === 'checkMask' || this.state === 'stunned' || this.state === 'attack' ||
   946→        (d < 9 && this.currentSpeed < 0.2);
   947→      if (interested && d < 13) {
   948→        _v1.subVectors(player.pos, this.pos);
   949→        const wantYaw = Math.atan2(_v1.x, _v1.z);
   950→        let diff = wantYaw - this.yaw;
   951→        while (diff > Math.PI) diff -= Math.PI * 2;
   952→        while (diff < -Math.PI) diff += Math.PI * 2;
   953→        lookYaw = Math.max(-1.0, Math.min(1.0, diff));
   954→        // mirar ligeramente hacia abajo: el guardia (1.6m) es más bajo
   955→        const headY = this.rig.height * 0.88;
   956→        const dHoriz = Math.max(0.4, Math.hypot(_v1.x, _v1.z));
   957→        lookPitch = Math.atan2(headY - 1.62, dHoriz);
   958→      }
   959→    }
   960→    const ctx: AnimCtx = {
   961→      dt,
   962→      time: t,
   963→      speed: this.currentSpeed,
   964→      walkRef: this.def.speedWalk,
   965→      state: (stateOverride ?? this.state) as AnimCtx['state'],
   966→      stateTime: this.stateTimer,
   967→      yaw: this.yaw,
   968→      crawling: this.crawling,
   969→      lookYaw,
   970→      lookPitch,
   971→    };
   972→    this.animator.update(ctx);
   973→
   974→    // pupilas más brillantes al cazar
   975→    const glow = this.state === 'chase' || this.state === 'hunt' || this.state === 'attack' ? 2.6 : 1.5;
   976→    for (const e of this.rig.eyes) {
   977→      const mats = Array.isArray(e.material) ? e.material : [e.material];
   978→      for (const m of mats as THREE.MeshStandardMaterial[]) {
   979→        m.emissiveIntensity = glow;
   980→      }
   981→    }
   982→    setEyeGlow(this.rig, this.active ? (glow > 2 ? 1 : 0.45) : 0.15);
   983→  }
   984→
   985→  dispose() {
   986→    this.rig.root.traverse((o) => {
