   253→  private buildWall(spec: WallSpec) {
   254→    const alongX = Math.abs(spec.z2 - spec.z1) < 0.001;
   255→    const a = alongX ? Math.min(spec.x1, spec.x2) : Math.min(spec.z1, spec.z2);
   256→    const b = alongX ? Math.max(spec.x1, spec.x2) : Math.max(spec.z1, spec.z2);
   257→    const fixed = alongX ? spec.z1 : spec.x1;
   258→    const H = spec.h;
   259→    const openings = (spec.openings ?? []).slice().sort((o1, o2) => o1.at - o2.at);
   260→
   261→    const mat = wallMaterials[spec.wallType]([Math.max(0.5, (b - a) / 2.4), H / 2.4]);
   262→
   263→    const solid = (from: number, to: number) => {
   264→      if (to - from < 0.05) return;
   265→      const len = to - from;
   266→      const cx = alongX ? (from + to) / 2 : fixed;
   267→      const cz = alongX ? fixed : (from + to) / 2;
   268→      const geo = new THREE.BoxGeometry(alongX ? len : WALL_T, H, alongX ? WALL_T : len);
   269→      const mesh = new THREE.Mesh(geo, mat);
   270→      mesh.position.set(cx, H / 2, cz);
   271→      mesh.castShadow = false;
   272→      mesh.receiveShadow = true;
   273→      this.root.add(mesh);
   274→      this.addCollider(
   275→        new THREE.Box3(
   276→          new THREE.Vector3(cx - (alongX ? len / 2 : WALL_T / 2 + 0.02), 0, cz - (alongX ? WALL_T / 2 + 0.02 : len / 2)),
   277→          new THREE.Vector3(cx + (alongX ? len / 2 : WALL_T / 2 + 0.02), H, cz + (alongX ? WALL_T / 2 + 0.02 : len / 2))
   278→        ),
   279→        undefined,
   280→        true
   281→      );
   282→    };
   283→
   284→    const lintel = (from: number, to: number, h: number) => {
   285→      if (H - h < 0.1) return;
   286→      const len = to - from;
   287→      const cx = alongX ? (from + to) / 2 : fixed;
   288→      const cz = alongX ? fixed : (from + to) / 2;
   289→      const geo = new THREE.BoxGeometry(alongX ? len : WALL_T, H - h, alongX ? WALL_T : len);
   290→      const mesh = new THREE.Mesh(geo, mat);
   291→      mesh.position.set(cx, h + (H - h) / 2, cz);
   292→      mesh.receiveShadow = true;
   293→      this.root.add(mesh);
   294→    };
   295→
   296→    let cursor = a;
   297→    for (const op of openings) {
   298→      const oStart = op.at - op.width / 2;
   299→      const oEnd = op.at + op.width / 2;
   300→      if (oStart > cursor + 0.02) solid(cursor, oStart);
   301→      lintel(oStart, oEnd, op.height ?? DOOR_H);
   302→      cursor = oEnd;
   303→    }
   304→    if (cursor < b - 0.02) solid(cursor, b);
   305→  }

   315→  private buildDoor(d: DoorDef) {
   316→    const g = new THREE.Group();
   317→    const isGate = d.id.startsWith('gate_');
   318→    const isMetal = !isGate && (
   319→      d.id.includes('electrical') || d.id.includes('generator') || d.id.includes('exit') ||
   320→      d.id.includes('workshop') || d.id.includes('loading') || d.id.includes('freezers') || d.id.includes('pump') ||
   321→      d.id.includes('storageN') || d.id.includes('storageS') || d.id.includes('techS') ||
   322→      d.id.includes('mega')
   323→    );
   324→    const w = d.width - 0.08;
   325→
   326→    if (isGate) {
   327→      // REJA DE SEGURIDAD de la guardería: barras verticales + marco
   328→      const barMat = metalMaterial([0.55, 1]);
   329→      const nBars = Math.max(6, Math.round(w / 0.24));
   330→      for (let i = 0; i < nBars; i++) {
   331→        const bx = (w / nBars) * (i + 0.5);
   332→        const bar = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 2.6, 8), barMat);
   333→        bar.position.set(bx, 1.3, 0);
   334→        bar.castShadow = true;
   335→        g.add(bar);
   336→      }
   337→      // travesaños horizontales
   338→      for (const hy of [0.45, 1.3, 2.15]) {
   339→        const cross = new THREE.Mesh(new THREE.BoxGeometry(w, 0.07, 0.07), barMat);
   340→        cross.position.set(w / 2, hy, 0);
   341→        g.add(cross);
   342→      }
   343→      // marco lateral
   344→      for (const sx of [0.02, w - 0.02]) {
   345→        const post = new THREE.Mesh(new THREE.BoxGeometry(0.09, 2.8, 0.11), darkMetalMaterial());
   346→        post.position.set(sx, 1.4, 0);
   347→        g.add(post);
   348→      }
   349→      // señalización de seguridad arriba
   350→      const sign = new THREE.Mesh(
   351→        new THREE.BoxGeometry(0.62, 0.17, 0.05),
   352→        new THREE.MeshStandardMaterial({ color: 0x3a1414, emissive: 0xd84830, emissiveIntensity: 0.8 })
   353→      );
   354→      sign.position.set(w / 2, 2.42, 0.05);
   355→      g.add(sign);
   356→    } else {
   357→      const leafMat = isMetal ? metalMaterial([0.6, 1]) : this.woodDoorMat();
   358→      const leaf = new THREE.Mesh(new THREE.BoxGeometry(w, 2.05, 0.07), leafMat);
   359→      leaf.position.set(w / 2, 1.02, 0);
   360→      leaf.castShadow = true;
   361→      g.add(leaf);
   362→      // Ventanilla en puertas de madera
   363→      if (!isMetal && !d.secret) {
   364→        const glass = new THREE.Mesh(
   365→          new THREE.BoxGeometry(w * 0.4, 0.5, 0.02),
   366→          new THREE.MeshPhysicalMaterial({ color: 0x181a1c, roughness: 0.15, metalness: 0.3, transparent: true, opacity: 0.85 })
   367→        );
   368→        glass.position.set(w / 2, 1.6, 0.04);
   369→        g.add(glass);
   370→        const frame = new THREE.Mesh(new THREE.BoxGeometry(w * 0.4 + 0.08, 0.58, 0.03), darkMetalMaterial());
   371→        frame.position.set(w / 2, 1.6, 0.02);
   372→        g.add(frame);
   373→      }
   374→      // Manilla
   375→      const handle = new THREE.Mesh(new THREE.BoxGeometry(0.18, 0.05, 0.05), darkMetalMaterial());
   376→      handle.position.set(w - 0.22, 1.02, 0.07);
   377→      g.add(handle);
   378→    }
   379→    if (d.secret) {
   380→      this.dressAsShelf(g, w);
   381→    }
   382→    if (d.id === 'door_exit') {
   383→      const bar = new THREE.Mesh(new THREE.BoxGeometry(w * 0.8, 0.08, 0.1), metalMaterial([0.5, 0.2]));
   384→      bar.position.set(w / 2, 1.05, 0.09);
   385→      g.add(bar);
   386→      const sign = new THREE.Mesh(
   387→        new THREE.BoxGeometry(0.5, 0.18, 0.06),
   388→        new THREE.MeshStandardMaterial({ color: 0x113311, emissive: 0x22ff44, emissiveIntensity: 0.9 })
   389→      );
   390→      sign.position.set(w / 2, 2.2, 0.06);
   391→      g.add(sign);
   392→    }
   393→
   394→    const alongX = d.axis === 'x'; // hueco en muro que corre en X
   395→    const hx = alongX ? d.x - d.width / 2 + 0.04 : d.x;
   396→    const hz = alongX ? d.z : d.z - d.width / 2 + 0.04;
   397→    if (alongX) {
   398→      g.position.set(hx, 0, d.z);
   399→    } else {
   400→      g.position.set(d.x, 0, hz);
   401→      g.rotation.y = Math.PI / 2;
   402→    }
   403→    let openDir = 1;
   404→    if (alongX) {
   405→      const rb = this.rooms.get(d.roomB);
   406→      openDir = rb && rb.def.rect[1] < d.z ? 1 : -1;
   407→    } else {
   408→      const rb = this.rooms.get(d.roomB);
   409→      openDir = rb && rb.def.rect[0] < d.x ? 1 : -1;
   410→    }
   411→
   412→    this.root.add(g);
   413→
   414→    const cx = d.x;
   415→    const cz = d.z;
   416→    const collider = this.addCollider(
   417→      new THREE.Box3(
   418→        new THREE.Vector3(cx - (alongX ? d.width / 2 : 0.1), 0, cz - (alongX ? 0.1 : d.width / 2)),
   419→        new THREE.Vector3(cx + (alongX ? d.width / 2 : 0.1), 2.1, cz + (alongX ? 0.1 : d.width / 2))
   420→      ),
   421→      `door_${d.id}`
   422→    );
   423→
   424→    const door: DoorRT = {
   425→      def: d,
   426→      group: g,
   427→      open: false,
   428→      targetAngle: 0,
   429→      angle: 0,
   430→      collider,
   431→      locked: !!d.locked,
   432→      code: d.code,
   433→      isSecret: !!d.secret,
   434→      opening: false,
   435→      lastToggle: 0,
   436→    };
   437→    (g as unknown as { userData: { door?: DoorRT } }).userData.door = door;
   438→    this.doors.set(d.id, door);
   439→  }

   466→  private buildVents() {
   467→    for (const v of this.map.vents as VentDef[]) {
   468→      const rt: VentRT = { def: v, coverClosed: false, coverTimer: 0, blocked: false, bangTimer: 0 };
   469→      const ax = v.a[0];
   470→      const az = v.a[1];
   471→      const bx = v.b[0];
   472→      const bz = v.b[1];
   473→      const dx = bx - ax;
   474→      const dz = bz - az;
   475→      const len = Math.hypot(dx, dz);
   476→      const alongX = Math.abs(dx) > Math.abs(dz);
   477→      const cx = (ax + bx) / 2;
   478→      const cz = (az + bz) / 2;
   479→      const ductMat = metalMaterial([Math.max(1, len / 1.2), 1]);
   480→
   481→      const ductLen = len + 0.5;
   482→      const floor = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 1.0, 0.04, alongX ? 1.0 : ductLen), ductMat);
   483→      floor.position.set(cx, 0.03, cz);
   484→      this.root.add(floor);
   485→      const sideA = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 0.08, 1.15, alongX ? 0.08 : ductLen), ductMat);
   486→      sideA.position.set(cx + (alongX ? 0 : 0.52), 0.575, cz + (alongX ? 0.52 : 0));
   487→      this.root.add(sideA);
   488→      const sideB = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 0.08, 1.15, alongX ? 0.08 : ductLen), ductMat);
   489→      sideB.position.set(cx - (alongX ? 0 : 0.52), 0.575, cz - (alongX ? 0.52 : 0));
   490→      this.root.add(sideB);
   491→      const top = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 1.0, 0.06, alongX ? 1.0 : ductLen), ductMat);
   492→      top.position.set(cx, 1.18, cz);
   493→      this.root.add(top);
   494→      this.addCollider(
   495→        new THREE.Box3().setFromCenterAndSize(sideA.position.clone().setY(0.6), new THREE.Vector3(alongX ? ductLen : 0.14, 1.2, alongX ? 0.14 : ductLen)),
   496→        `vent_side_${v.id}_a`
   497→      );
   498→      this.addCollider(
   499→        new THREE.Box3().setFromCenterAndSize(sideB.position.clone().setY(0.6), new THREE.Vector3(alongX ? ductLen : 0.14, 1.2, alongX ? 0.14 : ductLen)),
   500→        `vent_side_${v.id}_b`
   501→      );
   502→
   503→      this.buildGrate(ax, az, alongX);
   504→      this.buildGrate(bx, bz, alongX);
   505→
   506→      if (v.coverAt) {
   507→        const px = v.coverAt === 'a' ? ax : bx;
   508→        const pz = v.coverAt === 'a' ? az : bz;
   509→        const cover = new THREE.Mesh(
   510→          new THREE.BoxGeometry(alongX ? 1.0 : 0.08, 1.15, alongX ? 0.08 : 1.0),
   511→          new THREE.MeshStandardMaterial({ color: 0x565e66, roughness: 0.4, metalness: 0.85 })
   512→        );
   513→        cover.position.set(px + (alongX ? 0 : 0.2), 0.6, pz + (alongX ? 0.2 : 0));
   514→        cover.visible = false;
   515→        this.root.add(cover);
   516→        rt.coverMesh = cover;
   517→        rt.coverCollider = this.addCollider(
   518→          new THREE.Box3().setFromCenterAndSize(cover.position.clone(), new THREE.Vector3(alongX ? 1.0 : 0.2, 1.2, alongX ? 0.2 : 1.0)),
   519→          `vent_cover_${v.id}`
   520→        );
   521→        rt.coverCollider.active = false;
   522→      }
   523→      this.vents.set(v.id, rt);
   524→    }
   525→  }

   527→  private buildGrate(x: number, z: number, alongX: boolean) {
   528→    const g = new THREE.Group();
   529→    const barMat = darkMetalMaterial();
   530→    const n = 6;
   531→    for (let i = 0; i < n; i++) {
   532→      const bar = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.02, 1.1, 6), barMat);
   533→      const off = -0.42 + (i / (n - 1)) * 0.84;
   534→      bar.position.set(alongX ? 0 : off, 0.55, alongX ? off : 0);
   535→      g.add(bar);
   536→    }
   537→    const frame = new THREE.Mesh(new THREE.BoxGeometry(alongX ? 0.06 : 0.98, 1.2, alongX ? 0.98 : 0.06), barMat);
   538→    frame.position.set(0, 0.6, 0);
   539→    g.add(frame);
   540→    g.position.set(x, 0, z);
   541→    this.root.add(g);
   542→  }

   551→  private buildLights() {
   552→    const LM = 42;
   553→    for (const rt of this.rooms.values()) {
   554→      const amb = rt.def.ambientLight;
   555→      const h = rt.def.height ?? 3.6;
   556→      const dimFactor = rt.def.id === 'office' ? 0.6 : 0.17;
   557→      rt.lightBase = (amb?.intensity ?? 0.45) * LM * dimFactor;
   558→      rt.lightOn = LIGHTS_ON_DEFAULT.has(rt.def.id);
   559→      this.roomList.push(rt);
   560→      const fixture = new THREE.Mesh(
   561→        new THREE.BoxGeometry(1.1, 0.08, 0.4),
   562→        new THREE.MeshStandardMaterial({ color: 0x9a9482, emissive: new THREE.Color(rt.lightColor), emissiveIntensity: 0.02 })
   563→      );
   564→      fixture.position.set(rt.center.x, h - 0.06, rt.center.z);
   565→      this.root.add(fixture);
   566→      rt.fixture = fixture;
   567→    }
   568→
   569→    // ---- POOL: 5 puntuales de sala ----
   570→    for (let i = 0; i < 5; i++) {
   571→      const l = new THREE.PointLight(0xffd9a8, 0, 15, 1.7);
   572→      l.position.set(0, 3, 0);
   573→      this.root.add(l);
   574→      this.roomPool.push(l);
   575→    }
   576→
   577→    // ---- POOL: 1 roja de emergencia (sigue la más cercana) ----
   578→    const em = new THREE.PointLight(0xff2211, 0, 10, 1.9);
   579→    em.position.set(0, 3.4, 0);
   580→    this.root.add(em);
   581→    this.emergencyPool = em;
   582→    // luminarias rojas visibles en las 8 posiciones (meshes emisivos, baratos)
   583→    const emMat = new THREE.MeshStandardMaterial({ color: 0x330604, emissive: 0xff2211, emissiveIntensity: 1.6 });
   584→    for (const [x, z] of this.emPositions) {
   585→      const lamp = new THREE.Mesh(new THREE.BoxGeometry(0.34, 0.12, 0.2), emMat);
   586→      lamp.position.set(x, 3.55, z);
   587→      this.root.add(lamp);
   588→    }
   589→
   590→    // ---- POOL: 1 puntual de acento (cala / fríos / decorativas) ----
   591→    const ap = new THREE.PointLight(0xffffff, 0, 8, 1.9);
   592→    ap.position.set(0, 2.5, 0);
   593→    this.root.add(ap);
   594→    this.accentPointPool = ap;
   595→
   596→    // ---- POOL: 2 focos de acento (vitrinas / estatua / proyector) ----
   597→    for (let i = 0; i < 2; i++) {
   598→      const s = new THREE.SpotLight(0xffd9a8, 0, 10, 0.5, 0.5, 1.6);
   599→      s.position.set(0, 4, 0);
   600→      s.target.position.set(0, 0, 0);
   601→      this.root.add(s);
   602→      this.root.add(s.target);
   603→      this.accentSpotPool.push(s);
   604→    }
   605→
   606→    // ---- foco azul moribundo del escenario del atrio (fijo) ----
   607→    const stage = new THREE.SpotLight(0x5588ff, 9, 20, 0.5, 0.45, 1.2);
   608→    stage.position.set(0, 7.6, -8.2);
   609→    stage.target.position.set(0, 1.6, -6.2);
   610→    this.root.add(stage);
   611→    this.root.add(stage.target);
   612→    this.stageSpot = stage;
   613→  }

   630→  update(dt: number, focusPos: THREE.Vector3) {
   631→    this.clock += dt;
   632→    const t = this.clock;
   633→
   634→    // ---- selección de las 5 salas encendidas más cercanas (sin alocar) ----
   635→    // en modo cámara, la sala del feed entra SIEMPRE (impulso de visión nocturna)
   636→    const feed = this.camView ? this.roomAt(focusPos.x, focusPos.z) : null;
   637→    const chosen: RoomRT[] = [];
   638→    for (let slot = 0; slot < this.roomPool.length; slot++) {
   639→      let best: RoomRT | null = null;
   640→      let bestD = Infinity;
   641→      for (const r of this.roomList) {
   642→        if (!r.lightOn || chosen.includes(r)) continue;
   643→        const dx = r.center.x - focusPos.x;
   644→        const dz = r.center.z - focusPos.z;
   645→        let d = dx * dx + dz * dz;
   646→        if (r === feed) d *= 0.25; // prioridad a la sala del feed
   647→        if (d < bestD) { best = r; bestD = d; }
   648→      }
   649→      if (best) chosen.push(best);
   650→    }
   651→
   652→    // apagar las salas sin luz del pool y actualizar luminarias
   653→    for (const r of this.roomList) {
   654→      const slot = chosen.indexOf(r);
   655→      let intensity = 0;
   656→      if (slot >= 0) {
   657→        intensity = r.lightBase * this.power;
   658→        if (r.flicker > 0.02) {
   659→          const f =
   660→            0.72 +
   661→            0.28 * Math.sin(t * 7.3 + r.center.x) * Math.sin(t * 3.1 + r.center.z) +
   662→            (Math.random() < 0.004 * r.flicker * 60 ? -0.8 : 0);
   663→          intensity *= Math.max(0.05, f);
   664→        }
   665→        // impulso de la sala del feed de la tableta
   666→        if (r === feed) intensity = Math.max(intensity, r.lightBase * 0.55);
   667→        const l = this.roomPool[slot];
   668→        const h = r.def.height ?? 3.6;
   669→        l.position.set(r.center.x, h - 0.55, r.center.z);
   670→        l.color.setHex(r.lightColor);
   671→        l.intensity = intensity;
   672→        l.distance = r.def.id === 'atrium' ? 22 : 15;
   673→      }
   674→      r.lightLevel = intensity;
   675→      if (r.fixture) {
   676→        const fm = r.fixture.material as THREE.MeshStandardMaterial;
   677→        fm.emissiveIntensity = intensity > 2 ? Math.min(3, 0.8 + intensity / 26) : 0.02;
   678→      }
   679→    }
   680→
   681→    // ---- emergencia roja: la posición más cercana ----
   682→    if (this.emergencyPool) {
   683→      let bx = this.emPositions[0][0];
   684→      let bz = this.emPositions[0][1];
   685→      let bestD = Infinity;
   686→      for (const [x, z] of this.emPositions) {
   687→        const d = (x - focusPos.x) ** 2 + (z - focusPos.z) ** 2;
   688→        if (d < bestD) { bestD = d; bx = x; bz = z; }
   689→      }
   690→      this.emergencyPool.position.set(bx, 3.4, bz);
   691→      this.emergencyPool.intensity = 6;
   692→    }
   693→
   694→    // ---- acentos: el puntual y los 2 focos más cercanos ----
   695→    if (this.accentPointPool) {
   696→      let bestD = Infinity;
   697→      let best: AccentLightDef | null = null;
   698→      for (const a of this.accents) {
   699→        if (a.spot) continue;
   700→        const d = a.pos.distanceToSquared(focusPos);
   701→        if (d < bestD) { bestD = d; best = a; }
   702→      }
   703→      const l = this.accentPointPool;
   704→      if (best) {
   705→        l.position.copy(best.pos);
   706→        l.color.setHex(best.color);
   707→        l.intensity = best.intensity;
   708→        l.distance = best.distance;
   709→        l.decay = best.decay ?? 1.9;
   710→      } else {
   711→        l.intensity = 0;
   712→      }
   713→    }
   714→    for (const sp of this.accentSpotPool) {
   715→      let bestD = Infinity;
   716→      let best: AccentLightDef | null = null;
   717→      for (const a of this.accents) {
   718→        if (!a.spot) continue;
   719→        const d = a.pos.distanceToSquared(focusPos);
   720→        if (d < bestD) { bestD = d; best = a; }
   721→      }
   722→      if (best) {
   723→        sp.position.copy(best.pos);
   724→        sp.target.position.copy(best.target ?? best.pos);
   725→        sp.color.setHex(best.color);
   726→        sp.intensity = best.intensity;
   727→        sp.distance = best.distance;
   728→        sp.angle = best.angle ?? 0.5;
   729→        sp.penumbra = best.penumbra ?? 0.5;
   730→        sp.decay = best.decay ?? 1.6;
   731→      } else {
   732→        sp.intensity = 0;
   733→      }
   734→    }
   735→
   736→    for (const a of this.propAnims) {
   737→      a.update(dt, this.clock);
   738→    }
   739→  }

   741→  reset() {
   742→    for (const d of this.doors.values()) {
   743→      d.open = false;
   744→      d.targetAngle = 0;
   745→      d.angle = 0;
   746→      d.group.rotation.y = d.def.axis === 'x' ? 0 : Math.PI / 2;
   747→      d.collider.active = true;
   748→      d.locked = !!d.def.locked;
   749→      d.opening = false;
   750→    }
   751→    for (const v of this.vents.values()) {
   752→      v.coverClosed = false;
   753→      v.coverTimer = 0;
   754→      v.blocked = false;
   755→      if (v.coverMesh) v.coverMesh.visible = false;
   756→      if (v.coverCollider) v.coverCollider.active = false;
   757→    }
   758→    for (const r of this.rooms.values()) {
   759→      r.lightOn = LIGHTS_ON_DEFAULT.has(r.def.id);
   760→      r.blackoutTimer = 0;
   761→      r.lightLevel = 0;
   762→    }
   763→    for (const c of this.colliders) {
   764→      if (c.id?.startsWith('item_')) c.active = false;
   765→    }
   766→  }
