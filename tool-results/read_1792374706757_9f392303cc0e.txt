    86→    86→export function setEyeGlow(rig: { eyeGlow: THREE.Mesh | null; eyeGlows?: THREE.Mesh[] }, level: number) {
    87→    87→  const list = rig.eyeGlows && rig.eyeGlows.length ? rig.eyeGlows : rig.eyeGlow ? [rig.eyeGlow] : [];
    88→    88→  for (const e of list) {
    89→    89→    const m = e.material as THREE.MeshBasicMaterial;
    90→    90→    m.opacity = Math.min(1, 0.25 + level * 0.75);
    91→    91→    const base = (e.userData as { baseScale?: number }).baseScale ?? 1;
    92→    92→    e.scale.setScalar(base * (1 + level * 0.35));

   152→   152→  /** Construye el rig: GLTF del repo si está disponible; procedural si no */
   153→   153→  private buildRig(def: AnimatronicDef): ARig {
   154→   154→    const assetKind = def.kind; // 'freddy' | 'bonnie' | 'chica' | 'foxy' | 'golden' | 'luna'
   155→   155→    let gltfRig: GltfRig | null = null;
   156→   156→    if (assetKind === 'golden') {
   157→   157→      gltfRig = Assets.createVariant('golden', 0xc8a028);
   158→   158→    } else if (assetKind === 'luna') {
   159→   159→      // LUNA: modelo GLB estático del Eclipse (moon.glb) — vuela/rueda, no camina
   160→   160→      gltfRig = Assets.rigs.get('moon') ?? null;
   161→   161→    } else if (Assets.has(assetKind)) {
   162→   162→      gltfRig = Assets.rigs.get(assetKind) ?? null;
   163→   163→    }
   164→   164→    if (gltfRig) {
   165→   165→      /* v11 — HALOS DE OJOS CORRECTOS:
   166→   166→       * 1. Se muestrean los VÉRTICES SKINNED del mesh de ojos (applyBoneTransform),
   167→   167→       *    no la caja del nodo (que en Freddy cae bajo el suelo y en Bonnie se
   168→   168→       *    quedaba flotando quieta en la "bola morada").
   169→   169→       * 2. Si el mesh contiene ambos ojos, se separa en 2 grupos por X → un glow por ojo.
   170→   170→       * 3. Cada glow se ANCLA AL HUESO más cercano que gobierna esa zona
   171→   171→       *    (huesos de ojo si existen, si no el hueso cabeza) → sigue la
   172→   172→       *    animación (cabeceo, giro, persecución) en vez de quedarse flotando. */
   173→   173→      const eyeGlows: THREE.Mesh[] = [];
   174→   174→      const glowMat = new THREE.MeshBasicMaterial({
   175→   175→        color: def.eyeColor,
   176→   176→        transparent: true,
   177→   177→        opacity: 0.55,
   178→   178→        blending: THREE.AdditiveBlending,
   179→   179→        depthWrite: false,
   180→   180→        toneMapped: false,
   181→   181→      });
   182→   182→      gltfRig.root.updateMatrixWorld(true);
   183→   183→
   184→   184→      // huesos candidatos para anclar: ojos por nombre + cabeza
   185→   185→      const eyeBones: THREE.Object3D[] = [];
   186→   186→      gltfRig.model.traverse((o) => {
   187→   187→        if (!(o as THREE.Mesh).isMesh && /eye/i.test(o.name)) eyeBones.push(o);
   188→   188→      });
   189→   189→      const anchorCandidates: THREE.Object3D[] = [...eyeBones];
   190→   190→      if (gltfRig.bones.head) anchorCandidates.push(gltfRig.bones.head);
   191→   191→
   192→   192→      // muestrear centros de ojos a partir de vértices skinned (en mundo)
   193→   193→      const sampleEyeCenters = (mesh: THREE.Mesh): { centers: THREE.Vector3[]; radius: number } => {
   194→   194→        const pos = mesh.geometry.attributes.position as THREE.BufferAttribute | undefined;
   195→   195→        if (!pos) return { centers: [], radius: 0.03 };
   196→   196→        const v = new THREE.Vector3();
   197→   197→        const pts: THREE.Vector3[] = [];
   198→   198→        const step = Math.max(1, Math.floor(pos.count / 220));
   199→   199→        for (let i = 0; i < pos.count; i += step) {
   200→   200→          v.fromBufferAttribute(pos, i);
   201→   201→          if ((mesh as THREE.SkinnedMesh).isSkinnedMesh) {
   202→   202→            try { (mesh as THREE.SkinnedMesh).applyBoneTransform(i, v); } catch { /* noop */ }
   203→   203→          }
   204→   204→          mesh.localToWorld(v);
   205→   205→          pts.push(v.clone());
   206→   206→        }
   207→   207→        if (!pts.length) return { centers: [], radius: 0.03 };
   208→   208→        // caja real de los vértices muestreados
   209→   209→        const bb = new THREE.Box3().setFromPoints(pts);
   210→   210→        const size = bb.getSize(new THREE.Vector3());
   211→   211→        const radius = Math.max(0.024, Math.min(size.y, size.z) * 0.36);
   212→   212→        const split = size.x > size.y * 1.55; // ambos ojos en un solo mesh
   213→   213→        const centers: THREE.Vector3[] = [];
   214→   214→        if (split) {
   215→   215→          const midX = (bb.min.x + bb.max.x) / 2;
   216→   216→          for (const side of [-1, 1]) {
   217→   217→            const half = pts.filter((p) => (side < 0 ? p.x < midX : p.x >= midX));
   218→   218→            if (half.length) centers.push(new THREE.Box3().setFromPoints(half).getCenter(new THREE.Vector3()));
   219→   219→          }
   220→   220→        } else {
   221→   221→          centers.push(bb.getCenter(new THREE.Vector3()));
   222→   222→        }
   223→   223→        return { centers, radius };
   224→   224→      };
   225→   225→
   226→   226→      for (const eye of gltfRig.eyes) {
   227→   227→        const { centers, radius } = sampleEyeCenters(eye);
   228→   228→        for (const cWorld of centers) {
   229→   229→          // hueso de anclaje: el candidato más cercano al centro del ojo
   230→   230→          let anchor: THREE.Object3D = eye;
   231→   231→          let bestD = Infinity;
   232→   232→          for (const cand of anchorCandidates) {
   233→   233→            const cw = cand.getWorldPosition(new THREE.Vector3());
   234→   234→            const d = cw.distanceTo(cWorld);
   235→   235→            if (d < bestD && d < 0.9) { bestD = d; anchor = cand; }
   236→   236→          }
   237→   237→          const glow = new THREE.Mesh(new THREE.SphereGeometry(radius, 10, 8), glowMat);
   238→   238→          const inv = 1 / Math.max(0.0001, anchor.getWorldScale(new THREE.Vector3()).x);
   239→   239→          glow.position.copy(anchor.worldToLocal(cWorld.clone()));
   240→   240→          glow.scale.setScalar(inv);
   241→   241→          (glow.userData as { baseScale?: number }).baseScale = inv;
   242→   242→          anchor.add(glow);
   243→   243→          eyeGlows.push(glow);
   244→   244→        }
   245→   245→      }
   246→   246→      if (!eyeGlows.length && def.kind === 'luna') {
   247→   247→        // Luna (GLB sin meshes de ojo): brillo tenue de la cara, pegado al frente
   248→   248→        const glow = new THREE.Mesh(new THREE.SphereGeometry(0.075, 10, 8), glowMat);
   249→   249→        glow.position.set(0, gltfRig.height * 0.78, 0.12);
   250→   250→        gltfRig.root.add(glow);
   251→   251→        eyeGlows.push(glow);
   252→   252→      }
   253→   253→      const rig: ARig = {
   254→   254→        root: gltfRig.root,
   255→   255→        model: gltfRig.model,
   256→   256→        bones: gltfRig.bones,
   257→   257→        eyes: gltfRig.eyes,
   258→   258→        eyeGlows,
   259→   259→        eyeGlow: eyeGlows[0] ?? null,
   260→   260→        height: gltfRig.height,
   261→   261→      };
   262→   262→      this.moonRig = def.kind === 'luna' ? gltfRig : null;
   263→   263→      return rig;
   264→   264→    }
   265→   265→    // fallback procedural
   266→   266→    const char = buildAnimatronic(
   267→   267→      def.kind === 'freddy' || def.kind === 'golden' ? 'bear'
   268→   268→        : def.kind === 'bonnie' ? 'rabbit'
   269→   269→        : def.kind === 'chica' ? 'chicken'
   270→   270→        : def.kind === 'luna' ? 'solluna' : 'fox',
   271→   271→      def.eyeColor
   272→   272→    );
   273→   273→    const rig: ARig = {
   274→   274→      root: char.root,
   275→   275→      model: null,
   276→   276→      bones: char.bones,
   277→   277→      eyes: char.eyes,
   278→   278→      eyeGlows: char.eyeGlow ? [char.eyeGlow] : [],
   279→   279→      eyeGlow: char.eyeGlow ?? null,
   280→   280→      height: char.height,
   281→   281→    };
   282→   282→    this.charRig = char;
   283→   283→    return rig;
   284→   284→  }

   417→   417→
   418→   418→  update(dt: number, player: PlayerLike) {
   419→   419→    this.stateTimer += dt;
   420→   420→    this.thinkTimer -= dt;
   421→   421→    this.vocalTimer -= dt;
   422→   422→
   423→   423→    if (!this.active) {
   424→   424→      // retirado: ni caza ni despierta (Sol permanente)

   561→   561→  canSee(player: PlayerLike): boolean {
   562→   562→    if (player.hidden) return false;
   563→   563→    const d = this.pos.distanceTo(player.pos);
   564→   564→    let maxDist = this.def.visionDist * (0.75 + this.host.aggression * 0.35);
   565→   565→    const light = player.roomLightLevel();
   566→   566→    if (!player.flashlightOn && light < 0.25) maxDist *= 0.45;
   567→   567→    if (player.crouching) maxDist *= 0.8;
   568→   568→    if (player.flashlightOn) maxDist *= 1.25;
   569→   569→    if (d > maxDist) return false;
   570→   570→    if (this.def.visionFov < 360) {
   571→   571→      _v1.subVectors(player.pos, this.pos);
   572→   572→      _v1.y = 0;
   573→   573→      _v1.normalize();
   574→   574→      const facing = _v2.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   575→   575→      const dot = _v1.dot(facing);
   576→   576→      const cosHalf = Math.cos((this.def.visionFov * Math.PI) / 360);
   577→   577→      if (dot < cosHalf && d > 2.2) return false;
   578→   578→    }
   579→   579→    return this.host.nav.los(this.pos, player.pos, this.def.ventAccess);
   580→   580→  }

   836→   836→            this.path = p;
   837→   837→            this.pathIdx = 0;
   838→   838→            return;
   839→   839→          }
   840→   840→        }
   841→   841→      }
   842→   842→    }
   843→   843→  }
   844→   844→
   845→   845→  private arrived(r: number): boolean {
   846→   846→    if (!this.path.length) return true;
   847→   847→    const last = this.path[this.path.length - 1];
   848→   848→    return Math.hypot(this.pos.x - last.x, this.pos.z - last.z) < r;
   849→   849→  }
   850→   850→
   851→   851→  private followPath(dt: number, speed: number, player: PlayerLike) {
   852→   852→    if (!this.path.length || this.pathIdx >= this.path.length) {
   853→   853→      this.currentSpeed = Math.max(0, this.currentSpeed - dt * 6);
   854→   854→      return;
   855→   855→    }
   856→   856→    const target = this.path[this.pathIdx];
   857→   857→    _v1.set(target.x - this.pos.x, 0, target.z - this.pos.z);
   858→   858→    const d = _v1.length();
   859→   859→    if (d < 0.35) {
   860→   860→      this.pathIdx++;
   861→   861→      return;
   862→   862→    }
   863→   863→    _v1.normalize();
   864→   864→    const targetYaw = Math.atan2(_v1.x, _v1.z);
   865→   865→    let diff = targetYaw - this.yaw;
   866→   866→    while (diff > Math.PI) diff -= Math.PI * 2;
   867→   867→    while (diff < -Math.PI) diff += Math.PI * 2;
   868→   868→    this.yaw += Math.max(-4.5 * dt, Math.min(4.5 * dt, diff));
   869→   869→
   870→   870→    const cellVent = this.isVentCell(this.pos.x, this.pos.z);
   871→   871→    const targetVent = this.isVentCell(target.x, target.z);
   872→   872→    const wantCrawl = cellVent || targetVent;
   873→   873→    const crawlSpeedMul = wantCrawl ? 0.55 : 1;
   874→   874→    if (this.crawling !== wantCrawl) this.crawling = wantCrawl;
   875→   875→
   876→   876→    const v = speed * crawlSpeedMul;
   877→   877→    this.currentSpeed = v;
   878→   878→    const step = v * dt;
   879→   879→    this.pos.x += _v1.x * step;
   880→   880→    this.pos.z += _v1.z * step;
   881→   881→
   882→   882→    if (step < v * dt * 0.35) {
   883→   883→      this.stuckTimer += dt;
   884→   884→    } else {
   885→   885→      this.stuckTimer = Math.max(0, this.stuckTimer - dt);
   886→   886→    }
   887→   887→    if (this.stuckTimer > 0.8 && this.crawling) {
   888→   888→      this.host.sfx.ventBang(this.pos);
   889→   889→      this.stuckTimer = 0;
   890→   890→    }
   891→   891→
   892→   892→    this.autoDoors();
   893→   893→
   894→   894→    this.footTimer -= dt * (v > this.def.speedWalk * 1.05 ? 1.6 : 1);
   895→   895→    if (this.footTimer <= 0) {
   896→   896→      this.footTimer = 0.55 + Math.random() * 0.1;
   897→   897→      const surface = this.crawling ? 'metal' : this.surfaceAt();
   898→   898→      this.host.sfx.footstep(this.pos, surface, v > this.def.speedWalk * 1.05, true);
   899→   899→    }
   900→   900→  }
   901→   901→
   902→   902→  private isVentCell(x: number, z: number): boolean {
   903→   903→    const nav = this.host.nav;
   904→   904→    const i = nav.cellX(x);
   905→   905→    const j = nav.cellZ(z);
   906→   906→    if (i < 0 || j < 0 || i >= GRID_W || j >= GRID_H) return false;
   907→   907→    return nav.grid[j * GRID_W + i] === 2;
   908→   908→  }
   909→   909→
   910→   910→  private surfaceAt(): string {
   911→   911→    const room = this.host.world.roomAt(this.pos.x, this.pos.z);
   912→   912→    if (!room) return 'tile';
   913→   913→    switch (room.def.floor) {
   914→   914→      case 'wood': return 'wood';
   915→   915→      case 'metal': return 'metal';
   916→   916→      case 'carpet': return 'carpet';
   917→   917→      case 'concrete': return 'concrete';
   918→   918→      default: return 'tile';
   919→   919→    }
   920→   920→  }
   921→   921→
   922→   922→  private autoDoors() {
   923→   923→    for (const d of this.host.world.doors.values()) {
   924→   924→      if (d.open || d.locked) continue;
   925→   925→      const dd = Math.hypot(d.def.x - this.pos.x, d.def.z - this.pos.z);
   926→   926→      if (dd < 1.6) {
   927→   927→        d.open = true;
   928→   928→        d.targetAngle = d.group.rotation.y + (Math.random() > 0.5 ? 1.9 : -1.9);
   929→   929→        d.opening = true;
   930→   930→        this.host.sfx.doorCreak(this.pos);
   931→   931→      }
   932→   932→    }
   933→   933→  }

   935→   935→  /* ================== ANIMACIÓN (RigAnimator en espacio de personaje) ================== */
   936→   936→
   937→   937→  private updateAnimator(dt: number, player: PlayerLike | null, stateOverride?: string) {
   938→   938→    const t = this.host.time;
   939→   939→    let lookYaw = 0;
   940→   940→    let lookPitch = 0;
   941→   941→    if (player) {
   942→   942→      const d = this.pos.distanceTo(player.pos);
   943→   943→      const interested =
   944→   944→        stateOverride === 'dying' || this.state === 'chase' || this.state === 'hunt' ||
   945→   945→        this.state === 'checkMask' || this.state === 'stunned' || this.state === 'attack' ||
   946→   946→        (d < 9 && this.currentSpeed < 0.2);
   947→   947→      if (interested && d < 13) {
   948→   948→        _v1.subVectors(player.pos, this.pos);
   949→   949→        const wantYaw = Math.atan2(_v1.x, _v1.z);
   950→   950→        let diff = wantYaw - this.yaw;
   951→   951→        while (diff > Math.PI) diff -= Math.PI * 2;
   952→   952→        while (diff < -Math.PI) diff += Math.PI * 2;
   953→   953→        lookYaw = Math.max(-1.0, Math.min(1.0, diff));
   954→   954→        // mirar ligeramente hacia abajo: el guardia (1.6m) es más bajo
   955→   955→        const headY = this.rig.height * 0.88;
   956→   956→        const dHoriz = Math.max(0.4, Math.hypot(_v1.x, _v1.z));
   957→   957→        lookPitch = Math.atan2(headY - 1.62, dHoriz);
   958→   958→      }
   959→   959→    }
   960→   960→    const ctx: AnimCtx = {
   961→   961→      dt,
   962→   962→      time: t,
   963→   963→      speed: this.currentSpeed,
   964→   964→      walkRef: this.def.speedWalk,
   965→   965→      state: (stateOverride ?? this.state) as AnimCtx['state'],
   966→   966→      stateTime: this.stateTimer,
   967→   967→      yaw: this.yaw,
   968→   968→      crawling: this.crawling,
   969→   969→      lookYaw,
   970→   970→      lookPitch,
   971→   971→    };
   972→   972→    this.animator.update(ctx);
   973→   973→
   974→   974→    // pupilas más brillantes al cazar
   975→   975→    const glow = this.state === 'chase' || this.state === 'hunt' || this.state === 'attack' ? 2.6 : 1.5;
   976→   976→    for (const e of this.rig.eyes) {
   977→   977→      const mats = Array.isArray(e.material) ? e.material : [e.material];
   978→   978→      for (const m of mats as THREE.MeshStandardMaterial[]) {
   979→   979→        m.emissiveIntensity = glow;
   980→   980→      }
   981→   981→    }
   982→   982→    setEyeGlow(this.rig, this.active ? (glow > 2 ? 1 : 0.45) : 0.15);
   983→   983→  }
   984→   984→
   985→   985→  dispose() {
   986→   986→    this.rig.root.traverse((o) => {
   987→   987→      if (o instanceof THREE.Mesh) {
   988→   988→        o.geometry.dispose();
   989→   989→      }
   990→   990→    });
   991→   991→    this.host.world.scene.remove(this.rig.root);
   992→   992→  }
