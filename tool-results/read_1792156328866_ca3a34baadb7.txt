   118→  floorHeightAt(x: number, z: number): number {

   120→    if (x > 3.0 && x < 14.0 && z > -17.0 && z < -12.4) {
   121→      let f = 1;
   122→      if (z > -13.0) f = Math.max(0, (-13.0 - z) / 0.6 + 1); // rampa frontal
   123→      if (z > -12.4) f = 0;
   124→      return 0.45 * Math.min(1, Math.max(0, f));

   265→  private buildWall(spec: WallSpec) {
   266→    const alongX = Math.abs(spec.z2 - spec.z1) < 0.001;
   267→    const a = alongX ? Math.min(spec.x1, spec.x2) : Math.min(spec.z1, spec.z2);
   268→    const b = alongX ? Math.max(spec.x1, spec.x2) : Math.max(spec.z1, spec.z2);
   269→    const fixed = alongX ? spec.z1 : spec.x1;
   270→    const openings = (spec.openings ?? []).slice().sort((o1, o2) => o1.at - o2.at);
   271→
   272→    const mat = wallMaterials[spec.wallType]([Math.max(0.5, (b - a) / 2.4), WALL_H / 2.4]);
   273→
   274→    const solid = (from: number, to: number) => {
   275→      if (to - from < 0.05) return;
   276→      const len = to - from;
   277→      const cx = alongX ? (from + to) / 2 : fixed;
   278→      const cz = alongX ? fixed : (from + to) / 2;
   279→      const geo = new THREE.BoxGeometry(alongX ? len : WALL_T, WALL_H, alongX ? WALL_T : len);
   280→      const mesh = new THREE.Mesh(geo, mat);
   281→      mesh.position.set(cx, WALL_H / 2, cz);
   282→      mesh.castShadow = false;
   283→      mesh.receiveShadow = true;
   284→      this.root.add(mesh);
   285→      this.addCollider(
   286→        new THREE.Box3(
   287→          new THREE.Vector3(cx - (alongX ? len / 2 : WALL_T / 2 + 0.02), 0, cz - (alongX ? WALL_T / 2 + 0.02 : len / 2)),
   288→          new THREE.Vector3(cx + (alongX ? len / 2 : WALL_T / 2 + 0.02), WALL_H, cz + (alongX ? WALL_T / 2 + 0.02 : len / 2))
   289→        ),
   290→        undefined,
   291→        true
   292→      );
   293→    };
   294→
   295→    const lintel = (from: number, to: number, h: number) => {
   296→      if (WALL_H - h < 0.1) return;
   297→      const len = to - from;
   298→      const cx = alongX ? (from + to) / 2 : fixed;
   299→      const cz = alongX ? fixed : (from + to) / 2;
   300→      const geo = new THREE.BoxGeometry(alongX ? len : WALL_T, WALL_H - h, alongX ? WALL_T : len);
   301→      const mesh = new THREE.Mesh(geo, mat);
   302→      mesh.position.set(cx, h + (WALL_H - h) / 2, cz);
   303→      mesh.receiveShadow = true;
   304→      this.root.add(mesh);
   305→    };
   306→
   307→    let cursor = a;
   308→    for (const op of openings) {
   309→      const oStart = op.at - op.width / 2;
   310→      const oEnd = op.at + op.width / 2;
   311→      if (oStart > cursor + 0.02) solid(cursor, oStart);
   312→      lintel(oStart, oEnd, op.height ?? DOOR_H);
   313→      cursor = oEnd;
   314→    }
   315→    if (cursor < b - 0.02) solid(cursor, b);
   316→  }

   327→  private buildDoor(d: DoorDef) {
   328→    const g = new THREE.Group();
   329→    const isMetal = d.id.includes('electrical') || d.id.includes('machinery') || d.id.includes('exit');
   330→    const leafMat = isMetal ? metalMaterial([0.6, 1]) : this.woodDoorMat();
   331→    const w = d.width - 0.08;
   332→    const leaf = new THREE.Mesh(new THREE.BoxGeometry(w, 2.05, 0.07), leafMat);
   333→    leaf.position.set(w / 2, 1.02, 0);
   334→    leaf.castShadow = true;
   335→    g.add(leaf);
   336→    // Ventanilla en puertas de madera
   337→    if (!isMetal && !d.secret) {
   338→      const glass = new THREE.Mesh(
   339→        new THREE.BoxGeometry(w * 0.4, 0.5, 0.02),
   340→        new THREE.MeshPhysicalMaterial({ color: 0x181a1c, roughness: 0.15, metalness: 0.3, transparent: true, opacity: 0.85 })
   341→      );
   342→      glass.position.set(w / 2, 1.6, 0.04);
   343→      g.add(glass);
   344→      // Marco de la ventanilla
   345→      const frame = new THREE.Mesh(new THREE.BoxGeometry(w * 0.4 + 0.08, 0.58, 0.03), darkMetalMaterial());
   346→      frame.position.set(w / 2, 1.6, 0.02);
   347→      g.add(frame);
   348→    }
   349→    // Manilla
   350→    const handle = new THREE.Mesh(new THREE.BoxGeometry(0.18, 0.05, 0.05), darkMetalMaterial());
   351→    handle.position.set(w - 0.22, 1.02, 0.07);
   352→    g.add(handle);
   353→    if (d.secret) {
   354→      // Estantería falsa: cajas y tablas sobre la puerta
   355→      this.dressAsShelf(g, w);
   356→    }
   357→    if (d.id === 'door_exit') {
   358→      // Barra antipánico
   359→      const bar = new THREE.Mesh(new THREE.BoxGeometry(w * 0.8, 0.08, 0.1), metalMaterial([0.5, 0.2]));
   360→      bar.position.set(w / 2, 1.05, 0.09);
   361→      g.add(bar);
   362→      // Cartel SALIDA
   363→      const sign = new THREE.Mesh(
   364→        new THREE.BoxGeometry(0.5, 0.18, 0.06),
   365→        new THREE.MeshStandardMaterial({ color: 0x113311, emissive: 0x22ff44, emissiveIntensity: 0.9 })
   366→      );
   367→      sign.position.set(w / 2, 2.2, 0.06);
   368→      g.add(sign);
   369→    }
   370→
   371→    const alongX = d.axis === 'x'; // hueco en muro que corre en X
   372→    // bisagra en el extremo negativo del hueco
   373→    const hx = alongX ? d.x - d.width / 2 + 0.04 : d.x;
   374→    const hz = alongX ? d.z : d.z - d.width / 2 + 0.04;
   375→    if (alongX) {
   376→      g.position.set(hx, 0, d.z);
   377→    } else {
   378→      g.position.set(d.x, 0, hz);
   379→      g.rotation.y = Math.PI / 2;
   380→    }
   381→    // dirección de apertura (hacia roomB)
   382→    let openDir = 1;
   383→    if (alongX) {
   384→      const rb = this.rooms.get(d.roomB);
   385→      openDir = rb && rb.def.rect[1] < d.z ? 1 : -1; // abre hacia el sur si roomB está al sur
   386→    } else {
   387→      const rb = this.rooms.get(d.roomB);
   388→      openDir = rb && rb.def.rect[0] < d.x ? 1 : -1;
   389→    }
   390→
   391→    this.root.add(g);
   392→
   393→    const cx = alongX ? d.x : d.x;
   394→    const cz = alongX ? d.z : d.z;
   395→    const collider = this.addCollider(
   396→      new THREE.Box3(
   397→        new THREE.Vector3(cx - (alongX ? d.width / 2 : 0.1), 0, cz - (alongX ? 0.1 : d.width / 2)),
   398→        new THREE.Vector3(cx + (alongX ? d.width / 2 : 0.1), 2.1, cz + (alongX ? 0.1 : d.width / 2))
   399→      ),
   400→      `door_${d.id}`
   401→    );
   402→
   403→    const door: DoorRT = {
   404→      def: d,
   405→      group: g,
   406→      open: false,
   407→      targetAngle: 0,
   408→      angle: 0,
   409→      collider,
   410→      locked: !!d.locked,
   411→      code: d.code,
   412→      isSecret: !!d.secret,
   413→      opening: false,
   414→      lastToggle: 0,
   415→    };
   416→    (g as unknown as { userData: { door?: DoorRT } }).userData.door = door;
   417→    this.doors.set(d.id, door);
   418→  }

   424→  private dressAsShelf(g: THREE.Group, w: number) {
   425→    const shelfMat = woodDoorMatShelf();
   426→    for (let i = 0; i < 3; i++) {
   427→      const board = new THREE.Mesh(new THREE.BoxGeometry(w * 0.96, 0.05, 0.34), shelfMat);
   428→      board.position.set(w / 2, 0.5 + i * 0.62, 0.12);
   429→      g.add(board);
   430→      // cajas encima
   431→      for (let b = 0; b < 3; b++) {
   432→        const bw = 0.22 + Math.random() * 0.12;
   433→        const box = new THREE.Mesh(
   434→          new THREE.BoxGeometry(bw, 0.3 + Math.random() * 0.16, 0.22),
   435→          new THREE.MeshStandardMaterial({ color: 0x6b5a3e, roughness: 0.85 })
   436→        );
   437→        box.position.set(w * 0.18 + b * 0.3 + Math.random() * 0.06, 0.68 + i * 0.62, 0.12);
   438→        box.rotation.y = (Math.random() - 0.5) * 0.4;
   439→        g.add(box);
   440→      }
   441→    }
   442→  }

   446→  private buildVents() {
   447→    for (const v of this.map.vents as VentDef[]) {
   448→      const rt: VentRT = { def: v, coverClosed: false, coverTimer: 0, blocked: false, bangTimer: 0 };
   449→      // El conducto largo oeste se construye de extremo a extremo
   450→      const ax = v.a[0];
   451→      const az = v.a[1];
   452→      const bx = v.b[0];
   453→      const bz = v.b[1];
   454→      const dx = bx - ax;
   455→      const dz = bz - az;
   456→      const len = Math.hypot(dx, dz);
   457→      const alongX = Math.abs(dx) > Math.abs(dz);
   458→      const cx = (ax + bx) / 2;
   459→      const cz = (az + bz) / 2;
   460→      const ductMat = metalMaterial([Math.max(1, len / 1.2), 1]);
   461→
   462→      // Interior hueco: suelo, laterales y techo
   463→      const ductLen = len + 0.5;
   464→      const floor = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 1.0, 0.04, alongX ? 1.0 : ductLen), ductMat);
   465→      floor.position.set(cx, 0.03, cz);
   466→      this.root.add(floor);
   467→      const sideA = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 0.08, 1.15, alongX ? 0.08 : ductLen), ductMat);
   468→      sideA.position.set(cx + (alongX ? 0 : 0.52), 0.575, cz + (alongX ? 0.52 : 0));
   469→      this.root.add(sideA);
   470→      const sideB = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 0.08, 1.15, alongX ? 0.08 : ductLen), ductMat);
   471→      sideB.position.set(cx - (alongX ? 0 : 0.52), 0.575, cz - (alongX ? 0.52 : 0));
   472→      this.root.add(sideB);
   473→      const top = new THREE.Mesh(new THREE.BoxGeometry(alongX ? ductLen : 1.0, 0.06, alongX ? 1.0 : ductLen), ductMat);
   474→      top.position.set(cx, 1.18, cz);
   475→      this.root.add(top);
   476→      // Colliders de los laterales
   477→      this.addCollider(
   478→        new THREE.Box3().setFromCenterAndSize(sideA.position.clone().setY(0.6), new THREE.Vector3(alongX ? ductLen : 0.14, 1.2, alongX ? 0.14 : ductLen)),
   479→        `vent_side_${v.id}_a`
   480→      );
   481→      this.addCollider(
   482→        new THREE.Box3().setFromCenterAndSize(sideB.position.clone().setY(0.6), new THREE.Vector3(alongX ? ductLen : 0.14, 1.2, alongX ? 0.14 : ductLen)),
   483→        `vent_side_${v.id}_b`
   484→      );
   485→      if (len > 2.5) {
   486→        // relleno lateral para el conducto largo (impide caminar junto a él dentro del pasillo)
   487→        this.addCollider(
   488→          new THREE.Box3().setFromCenterAndSize(
   489→            new THREE.Vector3(cx + (alongX ? 0 : 1.0), 1.6, cz + (alongX ? 1.0 : 0)),
   490→            new THREE.Vector3(alongX ? ductLen : 0.6, 3.2, alongX ? 0.6 : ductLen)
   491→          ),
   492→          `vent_fill_${v.id}`
   493→        );
   494→        this.addCollider(
   495→          new THREE.Box3().setFromCenterAndSize(
   496→            new THREE.Vector3(cx - (alongX ? 0 : 1.0), 1.6, cz - (alongX ? 1.0 : 0)),
   497→            new THREE.Vector3(alongX ? ductLen : 0.6, 3.2, alongX ? 0.6 : ductLen)
   498→          ),
   499→          `vent_fill2_${v.id}`
   500→        );
   501→      }
   502→
   503→      // Rejillas en ambos extremos (barras)
   504→      this.buildGrate(ax, az, alongX);
   505→      this.buildGrate(bx, bz, alongX);
   506→
   507→      // Tapa si procede
   508→      if (v.coverAt) {
   509→        const px = v.coverAt === 'a' ? ax : bx;
   510→        const pz = v.coverAt === 'a' ? az : bz;
   511→        const cover = new THREE.Mesh(
   512→          new THREE.BoxGeometry(alongX ? 1.0 : 0.08, 1.15, alongX ? 0.08 : 1.0),
   513→          new THREE.MeshStandardMaterial({ color: 0x565e66, roughness: 0.4, metalness: 0.85 })
   514→        );
   515→        cover.position.set(px + (alongX ? 0 : 0.2), 0.6, pz + (alongX ? 0.2 : 0));
   516→        cover.visible = false;
   517→        this.root.add(cover);
   518→        rt.coverMesh = cover;
   519→        rt.coverCollider = this.addCollider(
   520→          new THREE.Box3().setFromCenterAndSize(cover.position.clone(), new THREE.Vector3(alongX ? 1.0 : 0.2, 1.2, alongX ? 0.2 : 1.0)),
   521→          `vent_cover_${v.id}`
   522→        );
   523→        rt.coverCollider.active = false;
   524→      }
   525→      this.vents.set(v.id, rt);
   526→    }
   527→  }

   529→  private buildGrate(x: number, z: number, alongX: boolean) {
   530→    const g = new THREE.Group();
   531→    const barMat = darkMetalMaterial();
   532→    const n = 6;
   533→    for (let i = 0; i < n; i++) {
   534→      const bar = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.02, 1.1, 6), barMat);
   535→      const off = -0.42 + (i / (n - 1)) * 0.84;
   536→      bar.position.set(alongX ? 0 : off, 0.55, alongX ? off : 0);
   537→      g.add(bar);
   538→    }
   539→    const frame = new THREE.Mesh(new THREE.BoxGeometry(alongX ? 0.06 : 0.98, 1.2, alongX ? 0.98 : 0.06), barMat);
   540→    frame.position.set(0, 0.6, 0);
   541→    g.add(frame);
   542→    g.position.set(x, 0, z);
   543→    this.root.add(g);
   544→  }

   548→  private buildLights() {
   549→    // Multiplicador de unidades físicas de Three.js r155+ (candelas)
   550→    const LM = 42;
   551→    for (const rt of this.rooms.values()) {
   552→      const amb = rt.def.ambientLight;
   553→      const color = amb?.color ?? 0xffd9a8;
   554→      // noche cerrada: la instalación está fundida en su mayoría; solo la
   555→      // oficina conserva luz tenue. Las salas "reparadas" encienden tenues.
   556→      const dimFactor = rt.def.id === 'office' ? 0.6 : 0.17;
   557→      const base = (amb?.intensity ?? 0.45) * LM * dimFactor;
   558→      const light = new THREE.PointLight(color, base, 15, 1.7);
   559→      light.position.set(rt.center.x, 2.85, rt.center.z);
   560→      this.root.add(light);
   561→      rt.light = light;
   562→      rt.lightBase = base;
   563→      rt.lightOn = LIGHTS_ON_DEFAULT.has(rt.def.id);
   564→      this.roomLights.push(rt);
   565→      // Luminaria empotrada
   566→      const fixture = new THREE.Mesh(
   567→        new THREE.BoxGeometry(1.1, 0.08, 0.4),
   568→        new THREE.MeshStandardMaterial({ color: 0x9a9482, emissive: new THREE.Color(color), emissiveIntensity: 1.4 })
   569→      );
   570→      fixture.position.set(rt.center.x, WALL_H - 0.06, rt.center.z);
   571→      this.root.add(fixture);
   572→      rt.fixture = fixture;
   573→    }
   574→
   575→    // Focos del escenario: solo queda un azul moribundo (siluetas)
   576→    const spotColors = [0xff5544, 0x44ff88, 0x5588ff];
   577→    const spotX = [5.5, 8.5, 11.5];
   578→    for (let i = 0; i < 3; i++) {
   579→      const spot = new THREE.SpotLight(spotColors[i], i === 2 ? 7 : 0, 18, 0.5, 0.45, 1.2);
   580→      spot.position.set(spotX[i], 3.0, -9.5);
   581→      spot.target.position.set(spotX[i], 1.2, -13.5);
   582→      spot.castShadow = i === 2;
   583→      this.root.add(spot);
   584→      this.root.add(spot.target);
   585→      this.stageSpots.push(spot);
   586→    }
   587→
   588→    // Luces de emergencia (rojas) — siempre encendidas, tenues
   589→    const emPos: Array<[number, number]> = [
   590→      [-19, 15], [18.5, 15.5], [0, 12.5], [-8.5, 1.5],
   591→    ];
   592→    for (const [x, z] of emPos) {
   593→      const l = new THREE.PointLight(0xff2211, 6, 9, 1.9);
   594→      l.position.set(x, 2.9, z);
   595→      this.root.add(l);
   596→      this.emergencyLights.push(l);
   597→    }
   598→  }

   604→  /** Parpadeo, culling de luces cercanas y animaciones de props.
   605→   * focus = posición de interés (jugador o cámara de seguridad activa) */
   606→  update(dt: number, focusPos: THREE.Vector3) {
   607→    this.clock += dt;
   608→    // Culling: solo las 11 luces más cercanas al foco
   609→    const sorted = this.roomLights
   610→      .map((r) => ({ r, d: r.center.distanceToSquared(focusPos) }))
   611→      .sort((a, b) => a.d - b.d);
   612→    for (let i = 0; i < sorted.length; i++) {
   613→      const { r } = sorted[i];
   614→      const enable = i < 11;
   615→      if (r.light) {
   616→        let intensity = 0;
   617→        if (enable && r.lightOn) {
   618→          intensity = r.lightBase * this.power;
   619→          // parpadeo
   620→          if (r.flicker > 0.02) {
   621→            const t = this.clock;
   622→            const f =
   623→              0.72 +
   624→              0.28 * Math.sin(t * 7.3 + r.center.x) * Math.sin(t * 3.1 + r.center.z) +
   625→              (Math.random() < 0.004 * r.flicker * 60 ? -0.8 : 0);
   626→            intensity *= Math.max(0.05, f);
   627→          }
   628→        }
   629→        r.light.intensity = intensity;
   630→        if (r.fixture) {
   631→          const fm = r.fixture.material as THREE.MeshStandardMaterial;
   632→          fm.emissiveIntensity = intensity > 2 ? Math.min(3, 0.8 + intensity / 26) : 0.02;
   633→        }
   634→      }
   635→    }
   636→    // modo tableta: la sala del feed recibe luz tenue aunque esté fundida
   637→    // (sensores de la cámara; solo afecta al render del feed)
   638→    if (this.camView) {
   639→      const r = this.roomAt(focusPos.x, focusPos.z);
   640→      if (r && r.light) {
   641→        const boost = r.lightBase * 0.55;
   642→        if (r.light.intensity < boost) r.light.intensity = boost;
   643→        if (r.fixture) {
   644→          const fm = r.fixture.material as THREE.MeshStandardMaterial;
   645→          fm.emissiveIntensity = 0.6;
   646→        }
   647→      }
   648→    }
   649→    // Animaciones de props (ventiladores, cortinas, pantallas)
   650→    for (const a of this.propAnims) {
   651→      a.update(dt, this.clock);
   652→    }
   653→  }

   655→  reset() {
   656→    // Puertas cerradas, tapas fuera, luces on
   657→    for (const d of this.doors.values()) {
   658→      d.open = false;
   659→      d.targetAngle = 0;
   660→      d.angle = 0;
   661→      d.group.rotation.y = d.def.axis === 'x' ? 0 : Math.PI / 2;
   662→      d.collider.active = true;
   663→      d.locked = !!d.def.locked;
   664→      d.opening = false;
   665→    }
   666→    for (const v of this.vents.values()) {
   667→      v.coverClosed = false;
   668→      v.coverTimer = 0;
   669→      v.blocked = false;
   670→      if (v.coverMesh) v.coverMesh.visible = false;
   671→      if (v.coverCollider) v.coverCollider.active = false;
   672→    }
   673→    for (const r of this.rooms.values()) {
   674→      r.lightOn = LIGHTS_ON_DEFAULT.has(r.def.id);
   675→      r.blackoutTimer = 0;
   676→    }
   677→    for (const c of this.colliders) {
   678→      if (c.id?.startsWith('item_')) c.active = false;
   679→    }
   680→  }
