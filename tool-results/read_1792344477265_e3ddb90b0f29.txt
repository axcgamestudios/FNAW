   104→  async init(onProgress: (frac: number, label: string) => void) {
   105→    onProgress(0.02, 'Starting 3D engines...');

   240→  private buildGuardArms() {
   241→    const skinTex = Assets.textures?.guardSkin;
   242→    const clothTex = Assets.textures?.guardUniform;
   243→    const group = new THREE.Group();
   244→
   245→    const skinMat = new THREE.MeshStandardMaterial({
   246→      map: skinTex ?? null, color: skinTex ? 0xffffff : 0xb08a68, roughness: 0.72, metalness: 0,
   247→    });
   248→    const clothMat = new THREE.MeshStandardMaterial({
   249→      map: clothTex ?? null, color: clothTex ? 0xffffff : 0x27292e, roughness: 0.88, metalness: 0,
   250→    });
   251→
   252→    /** miembro cilíndrico del punto A al B (espacio de cámara) */
   253→    const limb = (a: THREE.Vector3, b: THREE.Vector3, rA: number, rB: number, mat: THREE.Material) => {
   254→      const len = a.distanceTo(b);
   255→      const m = new THREE.Mesh(new THREE.CylinderGeometry(rA, rB, len, 10), mat);
   256→      m.position.copy(a).add(b).multiplyScalar(0.5);
   257→      m.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), b.clone().sub(a).normalize());
   258→      return m;
   259→    };
   260→
   261→    // ---- brazo derecho: antebrazo visible desde el borde inferior-derecho ----
   262→    // manga (uniforme) + puño remangado (piel) + mano
   263→    group.add(limb(new THREE.Vector3(0.5, -0.42, -0.1), new THREE.Vector3(0.38, -0.33, -0.24), 0.066, 0.056, clothMat));
   264→    group.add(limb(new THREE.Vector3(0.38, -0.33, -0.24), new THREE.Vector3(0.32, -0.29, -0.33), 0.052, 0.046, skinMat));
   265→    const handR = new THREE.Mesh(new THREE.BoxGeometry(0.052, 0.04, 0.09), skinMat);
   266→    handR.position.set(0.305, -0.26, -0.38);
   267→    handR.rotation.y = -0.15;
   268→    const thumbR = new THREE.Mesh(new THREE.BoxGeometry(0.019, 0.018, 0.048), skinMat);
   269→    thumbR.position.set(0.283, -0.242, -0.36);
   270→    group.add(handR, thumbR);
   271→
   272→    // ---- brazo izquierdo: sujeta el frente por debajo ----
   273→    group.add(limb(new THREE.Vector3(-0.36, -0.52, -0.12), new THREE.Vector3(-0.08, -0.42, -0.28), 0.064, 0.054, clothMat));
   274→    group.add(limb(new THREE.Vector3(-0.08, -0.42, -0.28), new THREE.Vector3(0.05, -0.37, -0.38), 0.05, 0.045, skinMat));
   275→    const handL = new THREE.Mesh(new THREE.BoxGeometry(0.054, 0.044, 0.088), skinMat);
   276→    handL.position.set(0.13, -0.33, -0.43);
   277→    handL.rotation.set(0.2, -0.5, 0);
   278→    const fingersL = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.034, 0.052), skinMat);
   279→    fingersL.position.set(0.18, -0.31, -0.48);
   280→    fingersL.rotation.set(0.2, -0.5, 0);
   281→    group.add(handL, fingersL);
   282→
   283→    group.traverse((o) => {
   284→      if (o instanceof THREE.Mesh) {
   285→        o.castShadow = false;
   286→        o.receiveShadow = false;
   287→        o.frustumCulled = false;
   288→      }
   289→    });
   290→    group.visible = false;
   291→    this.camera.add(group);
   292→    this.guardArms = group;
   293→  }

   295→  /** balanceo de los brazos en primera persona según el paso del jugador */
   296→  private updateGuardArms(dt: number) {
   297→    if (!this.guardArms || !this.guardArms.visible) return;
   298→    const t = performance.now() / 1000;
   299→    const speed = Math.hypot(this.player.vel.x, this.player.vel.z);
   300→    const bobY = Math.sin(t * speed * 2.3) * Math.min(0.02, speed * 0.006);
   301→    const bobX = Math.cos(t * speed * 1.15) * Math.min(0.014, speed * 0.004);
   302→    this.guardArms.position.set(bobX, bobY, 0);
   303→    const target = new THREE.Quaternion().setFromEuler(
   304→      new THREE.Euler(Math.sin(t * speed * 2.3) * 0.012 + Math.sin(t * 1.1) * 0.006, 0, Math.cos(t * speed * 1.15) * 0.01)
   305→    );
   306→    this.guardArmsSway.slerp(target, Math.min(1, dt * 8));
   307→    this.guardArms.quaternion.copy(this.guardArmsSway);
   308→    // lente de la linterna brilla al encender
   309→    if (this.flashlightLens) {
   310→      const m = this.flashlightLens.material as THREE.MeshStandardMaterial;
   311→      m.emissiveIntensity = this.player.flashlightOn ? 2.8 : 0.04;
   312→    }
   313→  }

   338→  private buildDust() {
   339→    const count = this.quality === 'low' ? 0 : this.quality === 'medium' ? 260 : 420;
   340→    if (count === 0) return;
   341→    const geo = new THREE.BufferGeometry();
   342→    const pos = new Float32Array(count * 3);
   343→    for (let i = 0; i < count; i++) {
   344→      pos[i * 3] = (Math.random() - 0.5) * 16;
   345→      pos[i * 3 + 1] = Math.random() * 3;
   346→      pos[i * 3 + 2] = (Math.random() - 0.5) * 16;
   347→    }
   348→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   349→    const mat = new THREE.PointsMaterial({
   350→      color: 0xbfae8a,
   351→      size: 0.018,
   352→      transparent: true,
   353→      opacity: 0.42,
   354→      depthWrite: false,
   355→      blending: THREE.AdditiveBlending,
   356→    });
   357→    this.dust = new THREE.Points(geo, mat);
   358→    this.scene.add(this.dust);
   359→  }

   361→  private updateDust(_dt: number) {
   362→    if (!this.dust) return;
   363→    this.dust.position.set(
   364→      Math.round(this.player.pos.x / 8) * 8,
   365→      0,
   366→      Math.round(this.player.pos.z / 8) * 8
   367→    );
   368→    const p = this.dust.geometry.getAttribute('position') as THREE.BufferAttribute;
   369→    const arr = p.array as Float32Array;
   370→    for (let i = 0; i < arr.length; i += 3) {
   371→      arr[i + 1] += Math.sin(this.time * 0.6 + i) * 0.0008 + 0.0012;
   372→      arr[i] += Math.cos(this.time * 0.4 + i * 0.7) * 0.0006;
   373→      if (arr[i + 1] > 3.1) arr[i + 1] = 0.05;
   374→    }
   375→    p.needsUpdate = true;
   376→  }

   410→  private menuCamera(dt: number) {
   411→    const t = performance.now() / 1000;
   412→    const CX = -37.5;
   413→    const CZ = 24.5;
   414→    // cámara fija frente al rostro de Freddy
   415→    this.camera.position.set(CX + Math.sin(t * 0.07) * 0.08, 2.18 + Math.sin(t * 0.11) * 0.02, CZ + 0.9);
   416→    this.camera.rotation.set(0, 0, 0);
   417→    this.camera.rotation.order = 'YXZ';
   418→    this.camera.rotation.y = 0 + Math.sin(t * 0.05) * 0.03;
   419→    this.camera.rotation.x = 0.1;
   420→    // espasmos de cabeza de Freddy
   421→    if (this.menuRig) {
   422→      this.menuTwitchT -= dt;
   423→      const head = this.menuRig.bones.head;
   424→      if (head) {
   425→        if (this.menuTwitchT <= 0) {
   426→          this.menuTwitchT = 1.2 + Math.random() * 3.5;
   427→        }
   428→        head.rotation.z = Math.sin(t * 0.8) * 0.03 + (Math.random() < 0.02 ? (Math.random() - 0.5) * 0.2 : 0);
   429→        head.rotation.y = Math.sin(t * 0.33) * 0.08;
   430→      }
   431→      for (const e of this.menuRig.eyes) {
   432→        const mats = Array.isArray(e.material) ? e.material : [e.material];
   433→        for (const m of mats as THREE.MeshStandardMaterial[]) {
   434→          m.emissiveIntensity = 1.2 + (Math.random() < 0.06 ? -1.1 : 0);
   435→        }
   436→      }

   723→  private onKeyDown(e: KeyboardEvent) {
   724→    if (e.code === 'Tab') e.preventDefault();
   725→    this.player.keys[e.code] = true;
   726→    if (this.mode === 'cinematic') {
   727→      if (e.code === 'KeyE' || e.code === 'Space' || e.code === 'Enter') {
   728→        e.preventDefault();
   729→        this.skipCinematic();
   730→      } else if (e.code === 'Escape') {
   731→        this.cine.stop(true);
   732→      }
   733→      return;
   734→    }
   735→    if (this.mode !== 'playing') {
   736→      if (e.code === 'Escape' && this.mode === 'paused') this.resume();
   737→      return;
   738→    }
   739→    switch (e.code) {
   740→      case 'KeyF':
   741→        if (!this.missions.hasFlashlight) {
   742→          this.notify('The flashlight is on the desk');
   743→          break;
   744→        }
   745→        this.player.toggleFlashlight();
   746→        // v11: CLICK REAL de linterna del repo (con fallback al blip)
   747→        if (!this.audio.flashlightClick(this.player.flashlightOn)) this.audio.camSwitch();
   748→        break;
   749→      case 'KeyQ':
   750→        this.toggleMask();
   751→        break;
   752→      case 'KeyC':
   753→        this.toggleTablet();
   754→        break;
   755→      case 'ControlLeft':
   756→      case 'ControlRight':
   757→        this.player.crouching = !this.player.crouching;
   758→        break;
   759→      case 'KeyE':
   760→        this.tryInteract();
   761→        break;
   762→      case 'Escape':
   763→        this.pause();
   764→        break;
   765→      case 'ArrowLeft':
   766→      case 'KeyA':
   767→        if (useHud.getState().tabletOpen) this.cams.switchCam(-1);
   768→        break;
   769→      case 'ArrowRight':
   770→      case 'KeyD':
   771→        if (useHud.getState().tabletOpen) this.cams.switchCam(1);
   772→        break;
   773→      default:
   774→        if (e.code.startsWith('Digit')) {
   775→          const n = parseInt(e.code.slice(5), 10);
   776→          const stt = useHud.getState();
   777→          if (stt.tabletOpen && n >= 1 && n <= 9) {
   778→            const cam = this.cams.cams.find((c) => c.index === n);
   779→            if (cam) this.cams.setCam(n);
   780→          }
   781→        }
   782→        break;
   783→    }
   784→  }
