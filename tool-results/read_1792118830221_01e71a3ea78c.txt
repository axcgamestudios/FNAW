    88→export class Animatronic {
    89→  def: AnimatronicDef;
    90→  rig: ARig;
    91→  host: AIHost;
    92→  state: AIState = 'dormant';
    93→  active = false;
    94→  scriptedAwake = false;
    95→  pos = new THREE.Vector3();
    96→  yaw = 0;
    97→  path: THREE.Vector3[] = [];
    98→  pathIdx = 0;
    99→  patrolTarget: THREE.Vector3 | null = null;
   100→  investigatePoint: THREE.Vector3 | null = null;
   101→  lastKnown: THREE.Vector3 | null = null;
   102→  lastSeenAt = -99;
   103→  stateTimer = 0;
   104→  thinkTimer = 0;
   105→  vocalTimer = 8;
   106→  footTimer = 0;
   107→  walkPhase = 0;
   108→  currentSpeed = 0;
   109→  lungeDir = new THREE.Vector3();
   110→  litTime = 0;
   111→  fooledTime = 0;
   112→  maskLocked = false;
   113→  twitchTimer = 0;
   114→  twitch = 0;
   115→  crawling = false;
   116→  stuckTimer = 0;
   117→  private rests: RestPose[] = [];
   118→  private mixerFade = 0;
   119→
   120→  constructor(def: AnimatronicDef, host: AIHost) {
   121→    this.def = def;
   122→    this.host = host;
   123→    this.rig = this.buildRig(def);
   124→    const room = host.world.rooms.get(def.startRoom);
   125→    const [ox, oz] = def.startOffset ?? [0, 0];
   126→    const cx = room ? room.center.x : 0;
   127→    const cz = room ? room.center.z : 0;
   128→    this.pos.set(cx + ox, 0, cz + oz);
   129→    this.rig.root.position.copy(this.pos);
   130→    this.rig.root.rotation.y = 0;
   131→    (this.rig.root as unknown as { userData: { animatronicId?: string } }).userData.animatronicId = def.id;
   132→    host.world.scene.add(this.rig.root);
   133→    // corregir T-pose: bajar los brazos insertando pivotes correctores
   134→    this.applyArmDownPose();
   135→  }
   136→
   137→  /** Construye el rig: GLTF del repo si está disponible; procedural si no */
   138→  private buildRig(def: AnimatronicDef): ARig {
   139→    const assetKind = def.kind; // 'freddy' | 'bonnie' | 'chica' | 'foxy' | 'golden'
   140→    let gltfRig: GltfRig | null = null;
   141→    if (assetKind === 'golden') {
   142→      gltfRig = Assets.createVariant('golden', 0xc8a028);
   143→    } else if (Assets.has(assetKind)) {
   144→      gltfRig = Assets.rigs.get(assetKind) ?? null;
   145→    }
   146→    if (gltfRig) {
   147→      const eyeGlow = new THREE.PointLight(def.eyeColor, 9, 5, 1.9);
   148→      eyeGlow.position.set(0, def.kind === 'golden' ? 1.8 : 1.6, 0.3);
   149→      gltfRig.root.add(eyeGlow);
   150→      const rig: ARig = {
   151→        root: gltfRig.root,
   152→        bones: gltfRig.bones,
   153→        eyes: gltfRig.eyes,
   154→        eyeGlow,
   155→        height: gltfRig.height,
   156→        mixer: gltfRig.mixer,
   157→        idleAction: gltfRig.idleAction,
   158→      };
   159→      this.storeRests(rig);
   160→      return rig;
   161→    }
   162→    // fallback procedural
   163→    const char = buildAnimatronic(
   164→      def.kind === 'freddy' || def.kind === 'golden' ? 'bear'
   165→        : def.kind === 'bonnie' ? 'rabbit'
   166→        : def.kind === 'chica' ? 'chicken' : 'fox',
   167→      def.eyeColor
   168→    );
   169→    const rig: ARig = {
   170→      root: char.root,
   171→      bones: char.bones,
   172→      eyes: char.eyes,
   173→      eyeGlow: char.eyeGlow ?? new THREE.PointLight(def.eyeColor, 0, 0.1),
   174→      height: char.height,
   175→      mixer: null,
   176→      idleAction: null,
   177→    };
   178→    this.storeRests(rig);
   179→    return rig;
   180→  }
   181→
   182→  private storeRests(rig: ARig) {
   183→    this.rests = [];
   184→    for (const bone of Object.values(rig.bones)) {
   185→      if (bone) {
   186→        this.rests.push({ bone, x: bone.rotation.x, y: bone.rotation.y, z: bone.rotation.z });
   187→      }
   188→    }
   189→  }
   190→
   191→  private rest(bone: THREE.Object3D | undefined): RestPose | undefined {
   192→    if (!bone) return undefined;
   193→    return this.rests.find((r) => r.bone === bone);
   194→  }
   195→
   196→  /** rotación aditiva sobre la pose de reposo */
   197→  private swing(bone: THREE.Object3D | undefined, dx: number, dy = 0, dz = 0) {
   198→    if (!bone) return;
   199→    const r = this.rest(bone);
   200→    bone.rotation.x = (r ? r.x : 0) + dx;
   201→    bone.rotation.y = (r ? r.y : 0) + dy;
   202→    bone.rotation.z = (r ? r.z : 0) + dz;
   203→  }
   204→
   205→  /**
   206→   * Corrige la T-pose de fábrica: inserta un pivote entre el hueso del hombro
   207→   * y su padre, y lo rota para que el brazo caiga (con leve apertura).
   208→   * El pivote preserva la transformación mundial original del hueso.
   209→   */
   210→  private applyArmDownPose() {
   211→    const rig = this.rig;
   212→    if (!rig.bones.armL && !rig.bones.armR) return;
   213→    this.rig.root.updateMatrixWorld(true);
   214→    const pairs: Array<[THREE.Object3D | undefined, number]> = [
   215→      [rig.bones.armL, 1],
   216→      [rig.bones.armR, -1],
   217→    ];
   218→    for (const [arm, side] of pairs) {
   219→      if (!arm || !arm.parent) continue;
   220→      // dirección mundial actual del brazo (hombro → codo)
   221→      const shoulder = arm.getWorldPosition(new THREE.Vector3());
   222→      const child = arm.children.find((c) => c.name && /forearm|lowerarm|elbow|lower/i.test(c.name)) ?? arm.children[0];
   223→      if (!child) continue;
   224→      const elbow = child.getWorldPosition(new THREE.Vector3());
   225→      const armDir = elbow.sub(shoulder).normalize();
   226→      if (armDir.y < -0.82) continue; // ya apunta abajo
   227→      // objetivo: hacia abajo con ligera apertura lateral (el lado _L del personaje está en +X)
   228→      const target = new THREE.Vector3(side * 0.2, -1, 0).normalize();
   229→      const Q = new THREE.Quaternion().setFromUnitVectors(armDir, target);
   230→      // insertar pivote que preserva la transformación local del hueso
   231→      const pivot = new THREE.Object3D();
   232→      pivot.name = arm.name + '_pivot';
   233→      pivot.position.copy(arm.position);
   234→      pivot.quaternion.copy(arm.quaternion);
   235→      pivot.scale.copy(arm.scale);
   236→      arm.parent.add(pivot);
   237→      arm.position.set(0, 0, 0);
   238→      arm.quaternion.identity();
   239→      arm.scale.set(1, 1, 1);
   240→      pivot.add(arm);
   241→      // rotación correctora conjugada al espacio local del padre
   242→      const pivotParent = pivot.parent!;
   243→      const parentQ = pivotParent.getWorldQuaternion(new THREE.Quaternion());
   244→      const qLocal = parentQ.clone().invert().multiply(Q).multiply(parentQ);
   245→      pivot.quaternion.copy(qLocal.multiply(pivot.quaternion));
   246→    }
   247→    // recalcular las poses de reposo tras la corrección
   248→    this.storeRests(rig);
   249→    this.rig.root.updateMatrixWorld(true);
   250→  }
   251→
   252→  reset() {
   253→    this.state = 'dormant';
   254→    this.active = false;
   255→    const room = this.host.world.rooms.get(this.def.startRoom);
   256→    const [ox, oz] = this.def.startOffset ?? [0, 0];
   257→    this.pos.set((room ? room.center.x : 0) + ox, 0, (room ? room.center.z : 0) + oz);
   258→    this.rig.root.position.copy(this.pos);
   259→    this.rig.root.rotation.set(0, 0, 0);
   260→    this.rig.root.visible = true;
   261→    this.path = [];
   262→    this.yaw = 0;
   263→    this.stateTimer = 0;
   264→    this.thinkTimer = 0;
   265→    this.crawling = false;
   266→    this.litTime = 0;
   267→    this.fooledTime = 0;
   268→    this.maskLocked = false;
   269→    this.currentSpeed = 0;
   270→    // restaurar pose de reposo
   271→    for (const r of this.rests) {
   272→      r.bone.rotation.set(r.x, r.y, r.z);
   273→    }
   274→    if (this.rig.idleAction) {
   275→      this.rig.idleAction.reset().play();
   276→      this.mixerFade = 1;
   277→    }
   278→  }
   279→
   280→  wake() {
   281→    if (this.active) return;
   282→    this.active = true;
   283→    this.state = 'patrol';
   284→    this.stateTimer = 0;
   285→    this.host.sfx.vocal(this.pos, this.def.voice);
   286→  }
   287→
   288→  /** activación guionizada (p. ej. Freddy Dorado al coger la cinta) */
   289→  scriptWake() {
   290→    this.scriptedAwake = true;
   291→    this.wake();
   292→    this.state = this.def.persistent ? 'hunt' : 'chase';
   293→    this.stateTimer = 0;
   294→    this.lastKnown = null;
   295→  }
   296→
   297→  get eyeColor(): number {
   298→    return this.def.eyeColor;
   299→  }

   301→  /* ================== ACTUALIZACIÓN PRINCIPAL ================== */
   302→
   303→  update(dt: number, player: PlayerLike) {
   304→    this.stateTimer += dt;
   305→    this.thinkTimer -= dt;
   306→    this.vocalTimer -= dt;
   307→    if (this.rig.mixer) this.rig.mixer.update(dt);
   308→
   309→    if (!this.active) {
   310→      // despertar por temporizador del capítulo
   311→      const delay = this.wakeDelayForChapter();
   312→      if (delay < 900 && this.host.chapterTime >= delay) {
   313→        this.wake();
   314→      } else {
   315→        this.animateIdle(dt, player);
   316→        return;
   317→      }
   318→    }
   319→
   320→    // vocalizaciones aleatorias
   321→    if (this.vocalTimer <= 0) {
   322→      this.vocalTimer = 14 + Math.random() * 22;
   323→      if (this.pos.distanceTo(player.pos) < 18) {
   324→        this.host.sfx.vocal(this.pos, this.def.voice);
   325→      }
   326→    }
   327→
   328→    // detección
   329→    if (this.thinkTimer <= 0) {
   330→      this.thinkTimer = 0.12 + Math.random() * 0.08;
   331→      this.think(player);
   332→    }
   333→
   334→    // linterna
   335→    this.flashlightCheck(dt, player);
   336→
   337→    // máquina de estados
   338→    switch (this.state) {
   339→      case 'patrol': this.updatePatrol(dt, player); break;
   340→      case 'investigate': this.updateInvestigate(dt, player); break;
   341→      case 'chase': this.updateChase(dt, player); break;
   342→      case 'hunt': this.updateHunt(dt, player); break;
   343→      case 'search': this.updateSearch(dt, player); break;
   344→      case 'checkMask': this.updateCheckMask(dt, player); break;
   345→      case 'stunned': this.updateStunned(dt); break;
   346→      case 'attack': this.updateAttack(dt, player); break;
   347→      case 'retreat': this.updateRetreat(dt, player); break;
   348→      default: this.animateIdle(dt, player); break;
   349→    }
   350→
   351→    // física mínima: y del suelo
   352→    const fy = this.host.world.floorHeightAt(this.pos.x, this.pos.z);
   353→    this.pos.y += (fy - this.pos.y) * Math.min(1, dt * 10);
   354→    this.rig.root.position.copy(this.pos);
   355→    if (!this.crawling) this.rig.root.rotation.set(0, this.yaw, 0);
   356→    else this.rig.root.rotation.set(0.55, this.yaw, 0.12);
   357→
   358→    this.animate(dt, player);
   359→  }
   360→
   361→  private wakeDelayForChapter(): number {
   362→    const c = Math.min(Math.max(this.host.chapter, 1), 4);
   363→    return this.def.wakeDelay?.[c - 1] ?? 0;
   364→  }
   365→
   366→  /* ================== DETECCIÓN ================== */
   367→
   368→  private think(player: PlayerLike) {
   369→    const d = this.pos.distanceTo(player.pos);
   370→    // ruido
   371→    if (this.state === 'patrol' || this.state === 'search' || this.state === 'retreat' || this.state === 'investigate') {
   372→      for (const n of this.host.noises) {
   373→        const age = this.host.time - n.worldTime;
   374→        if (age > 0 && age < 1.2) {
   375→          const dn = Math.hypot(this.pos.x - n.x, this.pos.z - n.z);
   376→          const radius = n.radius * this.def.hearing * (0.8 + this.host.aggression * 0.3);
   377→          if (dn < radius) {
   378→            this.investigatePoint = new THREE.Vector3(n.x, 0, n.z);
   379→            if (this.state !== 'investigate') {
   380→              this.state = 'investigate';
   381→              this.stateTimer = 0;
   382→              this.host.sfx.vocal(this.pos, this.def.voice);
   383→            }
   384→            return;
   385→          }
   386→        }
   387→      }
   388→    }
   389→
   390→    // visión
   391→    if (this.canSee(player)) {
   392→      this.lastKnown = player.pos.clone();
   393→      this.lastSeenAt = this.host.time;
   394→      if (this.state === 'hunt' || this.state === 'chase' || this.state === 'attack' || this.state === 'checkMask') {
   395→        // ya está sobre el jugador
   396→      } else {
   397→        const dist = this.pos.distanceTo(player.pos);
   398→        if (player.masked && this.def.maskFoolable && dist < 4.2 && dist > 1.2 && !this.maskLocked) {
   399→          this.state = 'checkMask';
   400→          this.stateTimer = 0;
   401→          this.fooledTime = 0;
   402→          this.host.sfx.servo(this.pos, this.def.voice);
   403→          return;
   404→        }
   405→        if (player.masked && !this.def.maskFoolable) {
   406→          this.host.sfx.roar(this.pos, this.def.voice);
   407→        }
   408→        this.startChase();
   409→      }
   410→    } else if (this.state === 'chase' && this.host.time - this.lastSeenAt > 2.6) {
   411→      this.state = 'search';
   412→      this.stateTimer = 0;
   413→    }
   414→  }
   415→
   416→  canSee(player: PlayerLike): boolean {
   417→    if (player.hidden) return false;
   418→    const d = this.pos.distanceTo(player.pos);
   419→    let maxDist = this.def.visionDist * (0.75 + this.host.aggression * 0.35);
   420→    const light = player.roomLightLevel();
   421→    if (!player.flashlightOn && light < 0.25) maxDist *= 0.45;
   422→    if (player.crouching) maxDist *= 0.8;
   423→    if (player.flashlightOn) maxDist *= 1.25;
   424→    if (d > maxDist) return false;
   425→    if (this.def.visionFov < 360) {
   426→      _v1.subVectors(player.pos, this.pos);
   427→      _v1.y = 0;
   428→      _v1.normalize();
   429→      const facing = _v2.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   430→      const dot = _v1.dot(facing);
   431→      const cosHalf = Math.cos((this.def.visionFov * Math.PI) / 360);
   432→      if (dot < cosHalf && d > 2.2) return false;
   433→    }
   434→    return this.host.nav.los(this.pos, player.pos, this.def.ventAccess);
   435→  }
   436→
   437→  /* ================== LINTERNA ================== */
   438→
   439→  private flashlightCheck(dt: number, player: PlayerLike) {
   440→    if (!player.flashlightOn) {
   441→      this.litTime = Math.max(0, this.litTime - dt);
   442→      return;
   443→    }
   444→    if (this.def.flashlight === 'none' || this.state === 'attack' || this.state === 'hunt') return;
   445→    const d = this.pos.distanceTo(player.pos);
   446→    if (d > 13) return;
   447→    _v1.subVectors(this.pos, player.pos).normalize();
   448→    if (_v1.dot(player.lookDir) < 0.955) return;
   449→    if (!this.host.nav.los(player.pos, this.pos, this.def.ventAccess)) return;
   450→    this.litTime += dt;
   451→    if (this.def.flashlight === 'freeze' && this.litTime > 0.1) {
   452→      if (this.state !== 'stunned' && this.state !== 'checkMask') {
   453→        this.state = 'stunned';
   454→        this.stateTimer = 0;
   455→        this.host.sfx.servo(this.pos, this.def.voice);
   456→      }
   457→    }
   458→    if (this.litTime > 2.4) {
   459→      this.litTime = 0;
   460→      this.retreat();
   461→      this.host.sfx.vocal(this.pos, this.def.voice);
   462→    }
   463→  }
   464→
   465→  /* ================== ESTADOS ================== */
   466→
   467→  private startChase() {
   468→    if (this.state !== 'chase' && this.state !== 'attack' && this.state !== 'checkMask') {
   469→      this.host.sfx.stinger();
   470→      this.host.sfx.roar(this.pos, this.def.voice);
   471→    }
   472→    this.state = 'chase';
   473→    this.stateTimer = 0;
   474→    this.repathToPlayer();
   475→  }
   476→
   477→  private updatePatrol(dt: number, player: PlayerLike) {
   478→    if (!this.patrolTarget || this.arrived(1.2) || this.stateTimer > 40) {
   479→      const rooms = this.def.patrolRooms;
   480→      const room = rooms[(Math.random() * rooms.length) | 0];
   481→      const r = this.host.world.rooms.get(room);
   482→      if (r) {
   483→        const [x1, z1, x2, z2] = r.def.rect;
   484→        let tx = x1 + 1 + Math.random() * (x2 - x1 - 2);
   485→        let tz = z1 + 1 + Math.random() * (z2 - z1 - 2);
   486→        if (Math.random() < 0.28 * this.host.aggression) {
   487→          const pr = this.host.world.roomAt(player.pos.x, player.pos.z);
   488→          if (pr && (rooms.includes(pr.def.id) || Math.random() < 0.3)) {
   489→            tx = player.pos.x + (Math.random() - 0.5) * 3;
   490→            tz = player.pos.z + (Math.random() - 0.5) * 3;
   491→          }
   492→        }
   493→        this.patrolTarget = new THREE.Vector3(tx, 0, tz);
   494→        this.setPath(this.patrolTarget);
   495→        this.stateTimer = 0;
   496→      }
   497→    }
   498→    this.followPath(dt, this.def.speedWalk * (0.85 + this.host.aggression * 0.25), player);
   499→  }
   500→
   501→  private updateInvestigate(dt: number, player: PlayerLike) {
   502→    if (this.investigatePoint) {
   503→      if (!this.path.length || this.arrived(1.4) || this.thinkTimer > 0.5) {
   504→        if (this.pos.distanceTo(this.investigatePoint) > 1.6 && this.stateTimer < 20) {
   505→          this.setPath(this.investigatePoint);
   506→        } else {
   507→          this.yaw += Math.sin(this.stateTimer * 2.2) * dt * 2.5;
   508→          if (this.stateTimer > 6) {
   509→            this.investigatePoint = null;
   510→            this.state = 'patrol';
   511→            this.stateTimer = 0;
   512→            this.patrolTarget = null;
   513→          }
   514→        }
   515→      }
   516→    } else {
   517→      this.state = 'patrol';
   518→    }
   519→    if (this.path.length) {
   520→      this.followPath(dt, this.def.speedWalk * 1.15, player);
   521→    } else {
   522→      this.animateIdle(dt, player);
   523→    }
   524→  }
   525→
   526→  private repathToPlayer() {
   527→    if (this.lastKnown) this.setPath(this.lastKnown.clone());
   528→  }
   529→
   530→  private updateChase(dt: number, player: PlayerLike) {
   531→    const repath = this.stateTimer > 1.6 || !this.path.length;
   532→    if (repath) {
   533→      this.stateTimer = 0.01;
   534→      if (this.canSee(player) || this.def.persistent) {
   535→        this.lastKnown = player.pos.clone();
   536→        this.lastSeenAt = this.host.time;
   537→      }
   538→      this.repathToPlayer();
   539→    }
   540→    const speed = this.def.speedRun * (0.9 + this.host.aggression * 0.18);
   541→    this.followPath(dt, speed, player);
   542→    const d = this.pos.distanceTo(player.pos);
   543→    if (d < 2.8 && d > 1.1 && player.masked && this.def.maskFoolable && !this.maskLocked) {
   544→      if (Math.random() < 0.55) {
   545→        this.state = 'checkMask';
   546→        this.stateTimer = 0;
   547→        this.fooledTime = 0;
   548→        this.host.sfx.servo(this.pos, this.def.voice);
   549→        return;
   550→      }
   551→      this.maskLocked = true;
   552→    }
   553→    if (d <= this.def.attackRange && !player.hidden) {
   554→      this.beginAttack(player);
   555→    }
   556→    if (this.stateTimer > 26) {
   557→      this.state = 'search';
   558→      this.stateTimer = 0;
   559→    }
   560→  }
   561→
   562→  private updateHunt(dt: number, player: PlayerLike) {
   563→    this.lastKnown = player.pos.clone();
   564→    if (this.stateTimer > 1.2 || !this.path.length) {
   565→      this.stateTimer = 0.01;
   566→      this.setPath(this.lastKnown);
   567→    }
   568→    const speed = this.def.speedRun * (0.85 + this.host.aggression * 0.2);
   569→    this.followPath(dt, speed, player);
   570→    const d = this.pos.distanceTo(player.pos);
   571→    if (d <= this.def.attackRange && !player.hidden) {
   572→      this.beginAttack(player);
   573→    }
   574→  }
   575→
   576→  private updateSearch(dt: number, player: PlayerLike) {
   577→    if (!this.lastKnown) {
   578→      this.state = 'patrol';
   579→      return;
   580→    }
   581→    if (!this.path.length || this.arrived(1.5)) {
   582→      const angle = Math.random() * Math.PI * 2;
   583→      const r = 2 + Math.random() * 5;
   584→      const target = new THREE.Vector3(
   585→        this.lastKnown.x + Math.cos(angle) * r,
   586→        0,
   587→        this.lastKnown.z + Math.sin(angle) * r
   588→      );
   589→      this.setPath(target);
   590→      if (this.stateTimer > 14) {
   591→        this.state = 'patrol';
   592→        this.stateTimer = 0;
   593→        this.patrolTarget = null;
   594→      }
   595→    }
   596→    this.followPath(dt, this.def.speedWalk * 1.3, player);
   597→  }
   598→
   599→  private updateCheckMask(dt: number, player: PlayerLike) {
   600→    this.currentSpeed = 0;
   601→    _v1.subVectors(player.pos, this.pos);
   602→    this.yaw = Math.atan2(_v1.x, _v1.z);
   603→    this.fooledTime += dt;
   604→    if (!player.masked || player.running) {
   605→      this.maskLocked = true;
   606→      this.startChase();
   607→      return;
   608→    }
   609→    if (this.fooledTime > this.def.maskFoolTime) {
   610→      this.retreat();
   611→    }
   612→  }
   613→
   614→  private updateStunned(dt: number) {
   615→    this.currentSpeed = 0;
   616→    if (this.stateTimer > 3.6) {
   617→      this.retreat();
   618→    }
   619→  }
   620→
   621→  private beginAttack(player: PlayerLike) {
   622→    this.state = 'attack';
   623→    this.stateTimer = 0;
   624→    this.currentSpeed = 0;
   625→    this.host.sfx.roar(this.pos, this.def.voice);
   626→    _v1.subVectors(player.pos, this.pos);
   627→    _v1.y = 0;
   628→    _v1.normalize();
   629→    this.lungeDir.copy(_v1);
   630→  }
   631→
   632→  private updateAttack(dt: number, player: PlayerLike) {
   633→    const windup = this.def.attackWindup;
   634→    if (this.stateTimer < windup) {
   635→      _v1.subVectors(player.pos, this.pos);
   636→      this.yaw = Math.atan2(_v1.x, _v1.z);
   637→      const d = this.pos.distanceTo(player.pos);
   638→      if (d > this.def.attackRange + 1.1) {
   639→        this.startChase();
   640→        return;
   641→      }
   642→      return;
   643→    }
   644→    const speed = 9;
   645→    this.pos.addScaledVector(this.lungeDir, speed * dt);
   646→    this.rig.root.position.copy(this.pos);
   647→    const d = this.pos.distanceTo(player.pos);
   648→    if (d < 0.85 || this.stateTimer > windup + 0.42) {
   649→      this.host.onDeath(this);
   650→    }
   651→  }
   652→
   653→  private retreat() {
   654→    this.state = 'retreat';
   655→    this.stateTimer = 0;
   656→    const rooms = this.def.patrolRooms;
   657→    const room = rooms[(Math.random() * rooms.length) | 0];
   658→    const r = this.host.world.rooms.get(room);
   659→    if (r) {
   660→      this.patrolTarget = new THREE.Vector3(r.center.x, 0, r.center.z);
   661→      this.setPath(this.patrolTarget);
   662→    }
   663→  }
   664→
   665→  private updateRetreat(dt: number, player: PlayerLike) {
   666→    if (!this.path.length || this.arrived(1.5) || this.stateTimer > 22) {
   667→      this.state = 'patrol';
   668→      this.stateTimer = 0;
   669→      this.patrolTarget = null;
   670→    } else {
   671→      this.followPath(dt, this.def.speedRun * 0.85, player);
   672→    }
   673→  }

   675→  /* ================== MOVIMIENTO ================== */

   677→  private setPath(target: THREE.Vector3) {
   678→    this.path = this.host.nav.findPath(this.pos, target, this.def.ventAccess);
   679→    this.pathIdx = 0;
   680→    this.stuckTimer = 0;
   681→    if (!this.path.length) {
   682→      const rooms = this.def.patrolRooms;
   683→      for (const roomId of rooms) {
   684→        const r = this.host.world.rooms.get(roomId);
   685→        if (r) {
   686→          const alt = new THREE.Vector3(r.center.x, 0, r.center.z);
   687→          const p = this.host.nav.findPath(this.pos, alt, this.def.ventAccess);
   688→          if (p.length) {
   689→            this.path = p;
   690→            this.pathIdx = 0;
   691→            return;
   692→          }
   693→        }
   694→      }
   695→    }
   696→  }
   697→
   698→  private arrived(r: number): boolean {
   699→    if (!this.path.length) return true;
   700→    const last = this.path[this.path.length - 1];
   701→    return Math.hypot(this.pos.x - last.x, this.pos.z - last.z) < r;
   702→  }
   703→
   704→  private followPath(dt: number, speed: number, player: PlayerLike) {
   705→    if (!this.path.length || this.pathIdx >= this.path.length) {
   706→      this.currentSpeed = Math.max(0, this.currentSpeed - dt * 6);
   707→      return;
   708→    }
   709→    const target = this.path[this.pathIdx];
   710→    _v1.set(target.x - this.pos.x, 0, target.z - this.pos.z);
   711→    const d = _v1.length();
   712→    if (d < 0.35) {
   713→      this.pathIdx++;
   714→      return;
   715→    }
   716→    _v1.normalize();
   717→    const targetYaw = Math.atan2(_v1.x, _v1.z);
   718→    let diff = targetYaw - this.yaw;
   719→    while (diff > Math.PI) diff -= Math.PI * 2;
   720→    while (diff < -Math.PI) diff += Math.PI * 2;
   721→    this.yaw += Math.max(-4.5 * dt, Math.min(4.5 * dt, diff));
   722→
   723→    const cellVent = this.isVentCell(this.pos.x, this.pos.z);
   724→    const targetVent = this.isVentCell(target.x, target.z);
   725→    const wantCrawl = cellVent || targetVent;
   726→    const crawlSpeedMul = wantCrawl ? 0.55 : 1;
   727→    if (this.crawling !== wantCrawl) this.crawling = wantCrawl;
   728→
   729→    const v = speed * crawlSpeedMul;
   730→    this.currentSpeed = v;
   731→    const step = v * dt;
   732→    this.pos.x += _v1.x * step;
   733→    this.pos.z += _v1.z * step;
   734→
   735→    if (step < v * dt * 0.35) {
   736→      this.stuckTimer += dt;
   737→    } else {
   738→      this.stuckTimer = Math.max(0, this.stuckTimer - dt);
   739→    }
   740→    if (this.stuckTimer > 0.8 && this.crawling) {
   741→      this.host.sfx.ventBang(this.pos);
   742→      this.stuckTimer = 0;
   743→    }
   744→
   745→    this.autoDoors();
   746→
   747→    this.footTimer -= dt * (v > this.def.speedWalk * 1.05 ? 1.6 : 1);
   748→    if (this.footTimer <= 0) {
   749→      this.footTimer = 0.55 + Math.random() * 0.1;
   750→      const surface = this.crawling ? 'metal' : this.surfaceAt();
   751→      this.host.sfx.footstep(this.pos, surface, v > this.def.speedWalk * 1.05, true);
   752→    }
   753→  }
   754→
   755→  private isVentCell(x: number, z: number): boolean {
   756→    const nav = this.host.nav;
   757→    const i = nav.cellX(x);
   758→    const j = nav.cellZ(z);
   759→    if (i < 0 || j < 0 || i >= GRID_W || j >= GRID_H) return false;
   760→    return nav.grid[j * GRID_W + i] === 2;
   761→  }
   762→
   763→  private surfaceAt(): string {
   764→    const room = this.host.world.roomAt(this.pos.x, this.pos.z);
   765→    if (!room) return 'tile';
   766→    switch (room.def.floor) {
   767→      case 'wood': return 'wood';
   768→      case 'metal': return 'metal';
   769→      case 'carpet': return 'carpet';
   770→      case 'concrete': return 'concrete';
   771→      default: return 'tile';
   772→    }
   773→  }
   774→
   775→  private autoDoors() {
   776→    for (const d of this.host.world.doors.values()) {
   777→      if (d.open || d.locked) continue;
   778→      const dd = Math.hypot(d.def.x - this.pos.x, d.def.z - this.pos.z);
   779→      if (dd < 1.6) {
   780→        d.open = true;
   781→        d.targetAngle = d.group.rotation.y + (Math.random() > 0.5 ? 1.9 : -1.9);
   782→        d.opening = true;
   783→        this.host.sfx.doorCreak(this.pos);
   784→      }
   785→    }
   786→  }

   788→  /* ================== ANIMACIÓN PROCEDURAL SOBRE EL ESQUELETO ================== */

   790→  private animateIdle(dt: number, player: PlayerLike) {
   791→    this.currentSpeed = Math.max(0, this.currentSpeed - dt * 4);
   792→    const t = this.host.time;
   793→    const b = this.rig.bones;
   794→    // mezcla del clip idle (si existe)
   795→    this.updateMixerFade(0.35);
   796→    // balanceo sutil
   797→    if (b.torso) this.swing(b.torso, 0, 0, Math.sin(t * 0.5 + this.def.eyeColor) * 0.012);
   798→    const d = this.pos.distanceTo(player.pos);
   799→    if (d < 9 && this.active) {
   800→      _v1.subVectors(player.pos, this.pos);
   801→      const wantYaw = Math.atan2(_v1.x, _v1.z);
   802→      let diff = wantYaw - this.yaw;
   803→      while (diff > Math.PI) diff -= Math.PI * 2;
   804→      while (diff < -Math.PI) diff += Math.PI * 2;
   805→      if (b.head) this.swing(b.head, 0, Math.max(-0.9, Math.min(0.9, diff)), 0);
   806→    } else if (b.head) {
   807→      this.swing(b.head, 0, 0, 0);
   808→    }
   809→    this.stepJaw(t);
   810→  }
   811→
   812→  /** control de mezcla entre clip idle y pose manual */
   813→  private updateMixerFade(target: number) {
   814→    if (!this.rig.idleAction) return;
   815→    this.mixerFade += (target - this.mixerFade) * 0.08;
   816→    this.rig.idleAction.setEffectiveWeight(this.mixerFade);
   817→    this.rig.idleAction.paused = this.mixerFade < 0.02;
   818→  }
   819→
   820→  private stepJaw(t: number) {
   821→    const b = this.rig.bones;
   822→    const open = this.state === 'attack' ? 0.5 : Math.max(0, Math.sin(t * 0.7 + this.def.eyeColor * 0.001)) * 0.045;
   823→    if (b.jaw) this.swing(b.jaw, open);
   824→  }

   826→  private animate(dt: number, player: PlayerLike) {
   827→    const b = this.rig.bones;
   828→    const t = this.host.time;
   829→
   830→    if (this.crawling) {
   831→      this.updateMixerFade(0);
   832→      this.walkPhase += dt * this.currentSpeed * 3.2;
   833→      const s = Math.sin(this.walkPhase);
   834→      this.swing(b.armL, -1.2 + s * 0.45);
   835→      this.swing(b.armR, -1.2 - s * 0.45);
   836→      this.swing(b.legL, 0.5 - s * 0.3);
   837→      this.swing(b.legR, 0.5 + s * 0.3);
   838→      if (b.torso) this.swing(b.torso, 0.18, 0, 0);
   839→      if (b.head) this.swing(b.head, -0.4, 0, 0);
   840→      return;
   841→    }
   842→
   843→    const moving = this.currentSpeed > 0.15;
   844→    if (moving) {
   845→      this.updateMixerFade(0);
   846→      const freq = this.currentSpeed * 2.4;
   847→      this.walkPhase += dt * freq;
   848→      const s = Math.sin(this.walkPhase);
   849→      const c = Math.cos(this.walkPhase);
   850→      const amp = Math.min(1, this.currentSpeed / this.def.speedWalk) * (this.currentSpeed > this.def.speedWalk * 1.05 ? 0.75 : 0.5);
   851→      this.swing(b.legL, s * 0.55 * amp);
   852→      this.swing(b.legR, -s * 0.55 * amp);
   853→      this.swing(b.shinL, Math.max(0, -s) * 0.7 * amp);
   854→      this.swing(b.shinR, Math.max(0, s) * 0.7 * amp);
   855→      this.swing(b.armL, -s * 0.45 * amp);
   856→      this.swing(b.armR, s * 0.45 * amp);
   857→      this.swing(b.forearmL, -0.3 - Math.max(0, s) * 0.35);
   858→      this.swing(b.forearmR, -0.3 - Math.max(0, -s) * 0.35);
   859→      if (b.pelvis) {
   860→        const r = this.rest(b.pelvis)!;
   861→        b.pelvis.position.y = (b.pelvis.position.y * 0.7 + (r.bone.position.y + Math.abs(c) * 0.03 * amp) * 0.3);
   862→      }
   863→      if (b.torso) this.swing(b.torso, this.currentSpeed > this.def.speedWalk * 1.05 ? 0.18 : 0.05, 0, s * 0.035 * amp);
   864→    } else {
   865→      this.walkPhase = 0;
   866→      this.updateMixerFade(0.35);
   867→      if (b.torso) this.swing(b.torso, 0, 0, Math.sin(t * 0.5) * 0.012);
   868→      this.swing(b.legL, 0);
   869→      this.swing(b.legR, 0);
   870→      this.swing(b.armL, -0.05 + Math.sin(t * 0.8) * 0.02);
   871→      this.swing(b.armR, -0.05 - Math.sin(t * 0.8) * 0.02);
   872→    }
   873→
   874→    if (this.state === 'attack') {
   875→      this.swing(b.armL, -2.1);
   876→      this.swing(b.armR, -2.1);
   877→      this.swing(b.forearmL, -0.45);
   878→      this.swing(b.forearmR, -0.45);
   879→      if (b.jaw) this.swing(b.jaw, 0.55);
   880→      if (b.torso) this.swing(b.torso, 0.3, 0, 0);
   881→    } else if (this.state === 'checkMask') {
   882→      if (b.head) this.swing(b.head, 0, 0, 0.24);
   883→      if (b.jaw) this.swing(b.jaw, 0.03);
   884→    } else if (this.state === 'stunned') {
   885→      if (b.head) this.swing(b.head, 0, 0, -0.18);
   886→      if (b.jaw) this.swing(b.jaw, 0.1);
   887→    } else {
   888→      this.stepJaw(t);
   889→    }
   890→
   891→    // mirar al jugador si está cerca
   892→    const d = this.pos.distanceTo(player.pos);
   893→    const interested = this.state === 'chase' || this.state === 'hunt' || this.state === 'checkMask' || this.state === 'stunned' || (d < 9 && !moving);
   894→    if (b.head && interested && d < 12) {
   895→      _v1.subVectors(player.pos, this.pos);
   896→      const wantYaw = Math.atan2(_v1.x, _v1.z);
   897→      let diff = wantYaw - this.yaw;
   898→      while (diff > Math.PI) diff -= Math.PI * 2;
   899→      while (diff < -Math.PI) diff += Math.PI * 2;
   900→      this.swing(b.head, 0, Math.max(-1.0, Math.min(1.0, diff)), 0);
   901→    } else if (b.head) {
   902→      this.swing(b.head, 0, 0, 0);
   903→    }
   904→
   905→    // espasmo aleatorio (terror)
   906→    this.twitchTimer -= dt;
   907→    if (this.twitchTimer <= 0) {
   908→      this.twitchTimer = 6 + Math.random() * 18;
   909→      this.twitch = 0.22;
   910→    }
   911→    if (this.twitch > 0 && b.head) {
   912→      this.twitch -= dt;
   913→      b.head.rotation.z += (Math.random() - 0.5) * 0.13;
   914→    }
   915→
   916→    // pupilas más brillantes al cazar
   917→    const glow = this.state === 'chase' || this.state === 'hunt' || this.state === 'attack' ? 2.6 : 1.5;
   918→    for (const e of this.rig.eyes) {
   919→      const mats = Array.isArray(e.material) ? e.material : [e.material];
   920→      for (const m of mats as THREE.MeshStandardMaterial[]) {
   921→        m.emissiveIntensity = glow;
   922→      }
   923→    }
   924→    if (this.rig.eyeGlow) this.rig.eyeGlow.intensity = this.state === 'dormant' ? 2 : glow * 4;
   925→  }
