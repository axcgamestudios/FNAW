    24→function box(
    25→  w: number, h: number, d: number, mat: THREE.Material,
    26→  x: number, y: number, z: number, ry = 0, collide = true, world?: WorldBuilder
    27→): THREE.Mesh {
    28→  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), mat);
    29→  m.position.set(x, y, z);
    30→  m.rotation.y = ry;
    31→  m.castShadow = true;
    32→  m.receiveShadow = true;
    33→  if (world) {
    34→    world.root.add(m);
    35→    if (collide) {
    36→      world.addCollider(
    37→        new THREE.Box3().setFromCenterAndSize(m.position.clone(), new THREE.Vector3(w, h, d)),
    38→        undefined,
    39→        h > 0.7
    40→      );
    41→    }
    42→  }
    43→  return m;
    44→}

    64→function ceilingFan(world: WorldBuilder, x: number, z: number) {
    65→  const g = new THREE.Group();
    66→  const motor = new THREE.Mesh(new THREE.CylinderGeometry(0.12, 0.14, 0.18, 10), darkMetalMaterial());
    67→  const rod = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 0.5, 6), darkMetalMaterial());
    68→  rod.position.y = 0.3;
    69→  g.add(motor, rod);
    70→  const blades = new THREE.Group();
    71→  for (let i = 0; i < 4; i++) {
    72→    const b = new THREE.Mesh(new THREE.BoxGeometry(0.9, 0.02, 0.16), plasticMaterial(0x8a7a58, 0.6));
    73→    b.position.x = 0.55;
    74→    const holder = new THREE.Group();
    75→    holder.add(b);
    76→    holder.rotation.y = (i / 4) * Math.PI * 2;
    77→    blades.add(holder);
    78→  }
    79→  blades.position.y = 0.55;
    80→  g.add(blades);
    81→  g.position.set(x, 3.2, z);
    82→  world.root.add(g);
    83→  FANS.push(blades);
    84→}

   103→function shelfUnit(world: WorldBuilder, x: number, z: number, ry = 0, len = 2.4, boxes = 6) {
   104→  const g = new THREE.Group();
   105→  const mat = darkMetalMaterial();
   106→  for (let i = 0; i < 3; i++) {
   107→    const board = new THREE.Mesh(new THREE.BoxGeometry(len, 0.05, 0.6), mat);
   108→    board.position.y = 0.35 + i * 0.65;
   109→    g.add(board);
   110→  }
   111→  for (const sx of [-len / 2 + 0.05, len / 2 - 0.05]) {
   112→    const post = new THREE.Mesh(new THREE.BoxGeometry(0.08, 1.95, 0.08), mat);
   113→    post.position.set(sx, 1.0, -0.28);
   114→    g.add(post);
   115→    const post2 = post.clone();
   116→    post2.position.z = 0.28;
   117→    g.add(post2);
   118→  }
   119→  // cajas de cartón
   120→  for (let i = 0; i < boxes; i++) {
   121→    const bw = 0.3 + Math.random() * 0.25;
   122→    const b = new THREE.Mesh(
   123→      new THREE.BoxGeometry(bw, 0.24 + Math.random() * 0.2, 0.3 + Math.random() * 0.2),
   124→      new THREE.MeshStandardMaterial({ color: 0x77664a, roughness: 0.9 })
   125→    );
   126→    b.position.set((Math.random() - 0.5) * (len - 0.5), 0.5 + ((i / boxes) * 2.6 | 0) * 0.65, (Math.random() - 0.5) * 0.3);
   127→    b.rotation.y = (Math.random() - 0.5) * 0.5;
   128→    g.add(b);
   129→  }
   130→  g.position.set(x, 0, z);
   131→  g.rotation.y = ry;
   132→  world.root.add(g);
   133→  world.addCollider(new THREE.Box3().setFromCenterAndSize(new THREE.Vector3(x, 1, z), new THREE.Vector3(ry % Math.PI < 0.1 ? len : 0.7, 2, ry % Math.PI < 0.1 ? 0.7 : len)));
   134→  return g;
   135→}

   138→function locker(world: WorldBuilder, x: number, z: number, ry: number) {
   139→  const g = new THREE.Group();
   140→  const mat = darkMetalMaterial();
   141→  const body = new THREE.Mesh(new THREE.BoxGeometry(0.8, 2.0, 0.6), mat);
   142→  body.position.y = 1.0;
   143→  body.castShadow = true;
   144→  g.add(body);
   145→  for (const dx of [-0.19, 0.19]) {
   146→    const doorP = new THREE.Mesh(new THREE.BoxGeometry(0.36, 1.9, 0.03), metalMaterial([0.3, 0.8]));
   147→    doorP.position.set(dx, 1.0, 0.32);
   148→    g.add(doorP);
   149→    const vent1 = new THREE.Mesh(new THREE.BoxGeometry(0.26, 0.03, 0.05), mat);
   150→    vent1.position.set(dx, 1.7, 0.34);
   151→    const vent2 = vent1.clone();
   152→    vent2.position.y = 1.6;
   153→    g.add(vent1, vent2);
   154→  }
   155→  g.position.set(x, 0, z);
   156→  g.rotation.y = ry;
   157→  world.root.add(g);
   158→  const c = Math.abs(Math.cos(ry)), s = Math.abs(Math.sin(ry));
   159→  world.addCollider(
   160→    new THREE.Box3().setFromCenterAndSize(new THREE.Vector3(x, 1, z), new THREE.Vector3(0.8 * c + 0.6 * s, 2, 0.6 * c + 0.8 * s)),
   161→    `locker_${x.toFixed(1)}_${z.toFixed(1)}`
   162→  );
   163→}

   165→function arcadeCabinet(world: WorldBuilder, x: number, z: number, ry: number, broken = false, screenColor = 0x37ff88) {
   166→  const g = new THREE.Group();
   167→  const bodyMat = plasticMaterial(broken ? 0x2a2226 : 0x3a2030, 0.55, 0.5);
   168→  const body = new THREE.Mesh(new THREE.BoxGeometry(0.85, 1.8, 0.9), bodyMat);
   169→  body.position.y = 0.9;
   170→  body.castShadow = true;
   171→  g.add(body);
   172→  const screenMat = new THREE.MeshStandardMaterial({
   173→    color: 0x050805,
   174→    emissive: broken ? 0x000000 : screenColor,
   175→    emissiveIntensity: 0,
   176→    roughness: 0.25,
   177→  });
   178→  const screen = new THREE.Mesh(new THREE.PlaneGeometry(0.62, 0.48), screenMat);
   179→  screen.position.set(0, 1.25, 0.44);
   180→  screen.rotation.x = -0.18;
   181→  g.add(screen);
   182→  if (!broken) {
   183→    const overlay = new THREE.Mesh(new THREE.PlaneGeometry(0.62, 0.48), new THREE.MeshStandardMaterial({ map: screenStaticTex(), emissive: 0x333333, emissiveMap: screenStaticTex(), emissiveIntensity: 1, color: 0x111111 }));
   184→    overlay.position.set(0, 1.25, 0.45);
   185→    overlay.rotation.x = -0.18;
   186→    g.add(overlay);
   187→    FLICKER_SCREENS.push(overlay);
   188→  }
   189→  const panel = new THREE.Mesh(new THREE.BoxGeometry(0.7, 0.08, 0.3), plasticMaterial(0x222222, 0.6));
   190→  panel.position.set(0, 0.95, 0.52);
   191→  panel.rotation.x = 0.25;
   192→  g.add(panel);
   193→  // marquesina
   194→  const marquee = new THREE.Mesh(
   195→    new THREE.BoxGeometry(0.8, 0.28, 0.14),
   196→    new THREE.MeshStandardMaterial({ color: 0x662233, emissive: 0xff2255, emissiveIntensity: broken ? 0 : 0.5 })
   197→  );
   198→  marquee.position.set(0, 1.82, 0.38);
   199→  g.add(marquee);
   200→  g.position.set(x, 0, z);
   201→  g.rotation.y = ry;
   202→  world.root.add(g);
   203→  const c = Math.abs(Math.cos(ry)), s = Math.abs(Math.sin(ry));
   204→  world.addCollider(
   205→    new THREE.Box3().setFromCenterAndSize(new THREE.Vector3(x, 0.9, z), new THREE.Vector3(0.9 * c + 0.95 * s, 1.9, 0.95 * c + 0.9 * s))
   206→  );
   207→  return g;
   208→}

   379→function buildStage(world: WorldBuilder) {

   392→  const curtainMat = new THREE.MeshStandardMaterial({ color: 0x5a1220, roughness: 0.9, bumpMap: woodMaterial([1, 1]).map, bumpScale: 0.08 });

   398→    (cur as unknown as { wavePhase: number }).wavePhase = cx;
   399→    world.propAnims.push({
   400→      update: (_dt, t) => {
   401→        cur.scale.x = 1 + Math.sin(t * 0.4 + cx) * 0.03;
   402→      },
   403→    });
   404→  }
   405→  // Fondo con estrellas
   406→  const backdrop = new THREE.Mesh(
   407→    new THREE.PlaneGeometry(9.2, 2.8),
   408→    new THREE.MeshStandardMaterial({ color: 0x0e1226, emissive: 0x223048, emissiveIntensity: 0.35, roughness: 0.6 })
   409→  );
   410→  backdrop.position.set(8.5, 1.4, -16.95);
