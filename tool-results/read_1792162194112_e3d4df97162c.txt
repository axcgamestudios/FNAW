   110→

   195→  private buildGuardArms() {
   196→    const skinTex = Assets.textures?.guardSkin;
   197→    const clothTex = Assets.textures?.guardUniform;
   198→    const group = new THREE.Group();
   199→
   200→    const skinMat = new THREE.MeshStandardMaterial({
   201→      map: skinTex ?? null, color: skinTex ? 0xffffff : 0xb08a68, roughness: 0.72, metalness: 0,
   202→    });
   203→    const clothMat = new THREE.MeshStandardMaterial({
   204→      map: clothTex ?? null, color: clothTex ? 0xffffff : 0x27292e, roughness: 0.88, metalness: 0,
   205→    });
   206→
   207→    /** miembro cilíndrico del punto A al B (espacio de cámara) */
   208→    const limb = (a: THREE.Vector3, b: THREE.Vector3, rA: number, rB: number, mat: THREE.Material) => {
   209→      const len = a.distanceTo(b);
   210→      const m = new THREE.Mesh(new THREE.CylinderGeometry(rA, rB, len, 10), mat);
   211→      m.position.copy(a).add(b).multiplyScalar(0.5);
   212→      m.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), b.clone().sub(a).normalize());
   213→      return m;
   214→    };
   215→
   216→    // ---- brazo derecho: antebrazo visible desde el borde inferior-derecho ----
   217→    // manga (uniforme) + puño remangado (piel) + mano
   218→    group.add(limb(new THREE.Vector3(0.5, -0.42, -0.1), new THREE.Vector3(0.38, -0.33, -0.24), 0.066, 0.056, clothMat));
   219→    group.add(limb(new THREE.Vector3(0.38, -0.33, -0.24), new THREE.Vector3(0.32, -0.29, -0.33), 0.052, 0.046, skinMat));
   220→    const handR = new THREE.Mesh(new THREE.BoxGeometry(0.052, 0.04, 0.09), skinMat);
   221→    handR.position.set(0.305, -0.26, -0.38);
   222→    handR.rotation.y = -0.15;
   223→    const thumbR = new THREE.Mesh(new THREE.BoxGeometry(0.019, 0.018, 0.048), skinMat);
   224→    thumbR.position.set(0.283, -0.242, -0.36);
   225→    group.add(handR, thumbR);
   226→
   227→    // ---- brazo izquierdo: sujeta el frente por debajo ----
   228→    group.add(limb(new THREE.Vector3(-0.36, -0.52, -0.12), new THREE.Vector3(-0.08, -0.42, -0.28), 0.064, 0.054, clothMat));
   229→    group.add(limb(new THREE.Vector3(-0.08, -0.42, -0.28), new THREE.Vector3(0.05, -0.37, -0.38), 0.05, 0.045, skinMat));
   230→    const handL = new THREE.Mesh(new THREE.BoxGeometry(0.054, 0.044, 0.088), skinMat);
   231→    handL.position.set(0.13, -0.33, -0.43);
   232→    handL.rotation.set(0.2, -0.5, 0);
   233→    const fingersL = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.034, 0.052), skinMat);
   234→    fingersL.position.set(0.18, -0.31, -0.48);
   235→    fingersL.rotation.set(0.2, -0.5, 0);
   236→    group.add(handL, fingersL);
   237→
   238→    group.traverse((o) => {
   239→      if (o instanceof THREE.Mesh) {
   240→        o.castShadow = false;
   241→        o.receiveShadow = false;
   242→        o.frustumCulled = false;
   243→      }
   244→    });
   245→    group.visible = false;
   246→    this.camera.add(group);
   247→    this.guardArms = group;
   248→  }

   250→  /** balanceo de los brazos en primera persona según el paso del jugador */
   251→  private updateGuardArms(dt: number) {
   252→    if (!this.guardArms || !this.guardArms.visible) return;
   253→    const t = performance.now() / 1000;
   254→    const speed = Math.hypot(this.player.vel.x, this.player.vel.z);
   255→    const bobY = Math.sin(t * speed * 2.3) * Math.min(0.02, speed * 0.006);
   256→    const bobX = Math.cos(t * speed * 1.15) * Math.min(0.014, speed * 0.004);
   257→    this.guardArms.position.set(bobX, bobY, 0);
   258→    const target = new THREE.Quaternion().setFromEuler(
   259→      new THREE.Euler(Math.sin(t * speed * 2.3) * 0.012 + Math.sin(t * 1.1) * 0.006, 0, Math.cos(t * speed * 1.15) * 0.01)
   260→    );
   261→    this.guardArmsSway.slerp(target, Math.min(1, dt * 8));
   262→    this.guardArms.quaternion.copy(this.guardArmsSway);
   263→    // lente de la linterna brilla al encender
   264→    if (this.flashlightLens) {
   265→      const m = this.flashlightLens.material as THREE.MeshStandardMaterial;
   266→      m.emissiveIntensity = this.player.flashlightOn ? 2.8 : 0.04;
   267→    }
   268→  }

   292→  private buildDust() {
   293→    const count = this.quality === 'low' ? 0 : this.quality === 'medium' ? 260 : 420;
   294→    if (count === 0) return;
   295→    const geo = new THREE.BufferGeometry();
   296→    const pos = new Float32Array(count * 3);
   297→    for (let i = 0; i < count; i++) {
   298→      pos[i * 3] = (Math.random() - 0.5) * 16;
   299→      pos[i * 3 + 1] = Math.random() * 3;
   300→      pos[i * 3 + 2] = (Math.random() - 0.5) * 16;
   301→    }
   302→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   303→    const mat = new THREE.PointsMaterial({
   304→      color: 0xbfae8a,
   305→      size: 0.018,
   306→      transparent: true,
   307→      opacity: 0.42,
   308→      depthWrite: false,
   309→      blending: THREE.AdditiveBlending,
   310→    });
   311→    this.dust = new THREE.Points(geo, mat);
   312→    this.scene.add(this.dust);
   313→  }

   315→  private updateDust(_dt: number) {
   316→    if (!this.dust) return;
   317→    this.dust.position.set(
   318→      Math.round(this.player.pos.x / 8) * 8,
   319→      0,
   320→      Math.round(this.player.pos.z / 8) * 8
   321→    );
   322→    const p = this.dust.geometry.getAttribute('position') as THREE.BufferAttribute;
   323→    const arr = p.array as Float32Array;
   324→    for (let i = 0; i < arr.length; i += 3) {
   325→      arr[i + 1] += Math.sin(this.time * 0.6 + i) * 0.0008 + 0.0012;
   326→      arr[i] += Math.cos(this.time * 0.4 + i * 0.7) * 0.0006;
   327→      if (arr[i + 1] > 3.1) arr[i + 1] = 0.05;
   328→    }
   329→    p.needsUpdate = true;
   330→  }

   365→  private menuCamera(dt: number) {
   366→    const t = performance.now() / 1000;
   367→    const CX = -37.5;
   368→    const CZ = 24.5;
   369→    // cámara fija frente al rostro de Freddy
   370→    this.camera.position.set(CX + Math.sin(t * 0.07) * 0.08, 2.18 + Math.sin(t * 0.11) * 0.02, CZ + 0.9);
   371→    this.camera.rotation.set(0, 0, 0);
   372→    this.camera.rotation.order = 'YXZ';
   373→    this.camera.rotation.y = 0 + Math.sin(t * 0.05) * 0.03;
   374→    this.camera.rotation.x = 0.1;
   375→    // espasmos de cabeza de Freddy
   376→    if (this.menuRig) {
   377→      this.menuTwitchT -= dt;
   378→      const head = this.menuRig.bones.head;
   379→      if (head) {
   380→        if (this.menuTwitchT <= 0) {
   381→          this.menuTwitchT = 1.2 + Math.random() * 3.5;
   382→        }
   383→        head.rotation.z = Math.sin(t * 0.8) * 0.03 + (Math.random() < 0.02 ? (Math.random() - 0.5) * 0.2 : 0);
   384→        head.rotation.y = Math.sin(t * 0.33) * 0.08;
   385→      }
   386→      for (const e of this.menuRig.eyes) {
   387→        const mats = Array.isArray(e.material) ? e.material : [e.material];
   388→        for (const m of mats as THREE.MeshStandardMaterial[]) {
   389→          m.emissiveIntensity = 1.2 + (Math.random() < 0.06 ? -1.1 : 0);
   390→        }
   391→      }
   392→      this.menuRig.mixer?.update(dt);
   393→    }
   394→    if (this.world) this.world.update(dt, this.camera.position);
   395→  }

   399→  startChapter(n: number) {
   400→    this.chapter = n;
   401→    const chapters = (chaptersData as { chapters: ChapterDef[] }).chapters;
   402→    this.chapterDef = chapters.find((x) => x.chapter === n) ?? chapters[0];
   403→    this.time = 0;
   404→    this.power = this.chapterDef.startPower;
   405→    this.noises = [];
   406→    this.danger = 0;
   407→    this.chapterEndPending = false;
   408→    this.powerOut = false;
   409→    // quitar Freddy del menú y su foco
   410→    if (this.menuRig) this.scene.remove(this.menuRig.root);
   411→    if (this.menuSpot) this.menuSpot.intensity = 0;
   412→    // ocultar linterna/brazos hasta recogerla de nuevo
   413→    if (this.flashlightModel) this.flashlightModel.visible = false;
   414→    if (this.guardArms) this.guardArms.visible = false;
   415→    this.world.reset();
   416→    this.player.reset();
   417→    for (const a of this.animatronics) a.reset();
   418→    this.cams.reset(n);
   419→    this.missions = new MissionSystem(this.chapterDef, this.world, useHud, this.makeMissionHost());
   420→    this.events = new EventDirector(this.makeEventHost());
   421→    this.trials.reset(n);
   422→    this.buildInteractables();
   423→    this.trialInteractables = [];
   424→    this.callQueue = [];
   425→    this.callLineIdx = 0;
   426→    this.callLineTimer = 0;
   427→    useHud.getState().syncHud({
   428→      chapter: n,
   429→      chapterTitle: this.chapterDef.title,
   430→      hour: 0,
   431→      power: this.power,
   432→      objectives: [],
   433→      prompt: null,
   434→      subtitle: null,
   435→      deadBy: null,
   436→      chapterComplete: false,
   437→      ending: null,
   438→      jumpscare: null,
   439→    });
   440→    this.missions.syncObjectives();
   441→    this.audio.startAmbient();
   442→    this.audio.setChase(false);
   443→    this.audio.resume();
   444→    // ¿cinemática de entrada?
   445→    if (this.chapterDef.introCinematic) {
   446→      this.playCinematic(this.chapterDef.introCinematic, () => {
   447→        this.beginPlay();
   448→      });
   449→    } else {
   450→      this.beginPlay();
   451→    }
   452→  }

   598→  private onKeyDown(e: KeyboardEvent) {
   599→    if (e.code === 'Tab') e.preventDefault();
   600→    this.player.keys[e.code] = true;
   601→    if (this.mode === 'cinematic') {
   602→      if (e.code === 'KeyE' || e.code === 'Space' || e.code === 'Enter') {
   603→        e.preventDefault();
   604→        this.skipCinematic();
   605→      } else if (e.code === 'Escape') {
   606→        this.cine.stop(true);
   607→      }
   608→      return;
   609→    }
   610→    if (this.mode !== 'playing') {
   611→      if (e.code === 'Escape' && this.mode === 'paused') this.resume();
   612→      return;
   613→    }
   614→    switch (e.code) {
   615→      case 'KeyF':
   616→        if (!this.missions.hasFlashlight) {
   617→          this.notify('La linterna está sobre el escritorio');
   618→          break;
   619→        }
   620→        this.player.toggleFlashlight();
   621→        this.audio.camSwitch();
   622→        break;
   623→      case 'KeyQ':
   624→        this.toggleMask();
   625→        break;
   626→      case 'KeyC':
   627→        this.toggleTablet();
   628→        break;
   629→      case 'ControlLeft':
   630→      case 'ControlRight':
   631→        this.player.crouching = !this.player.crouching;
   632→        break;
   633→      case 'KeyE':
   634→        this.tryInteract();
   635→        break;
   636→      case 'Escape':
   637→        this.pause();
   638→        break;
   639→      case 'ArrowLeft':
   640→      case 'KeyA':
   641→        if (useHud.getState().tabletOpen) this.cams.switchCam(-1);
   642→        break;
   643→      case 'ArrowRight':
   644→      case 'KeyD':
   645→        if (useHud.getState().tabletOpen) this.cams.switchCam(1);
   646→        break;
   647→      default:
   648→        if (e.code.startsWith('Digit')) {
   649→          const n = parseInt(e.code.slice(5), 10);
   650→          const stt = useHud.getState();
   651→          if (stt.tabletOpen && n >= 1 && n <= 9) {
   652→            const cam = this.cams.cams.find((c) => c.index === n);
   653→            if (cam) this.cams.setCam(n);
   654→          }
   655→        }
   656→        break;
   657→    }
   658→  }

   661→    this.player.keys[e.code] = false;
   662→    if (e.code === 'KeyE') this.holdInteract = null;
   663→  }

   918→  private updateInteractions(dt: number) {
   919→    const st = useHud.getState();
   920→    if (this.mode !== 'playing' || st.tabletOpen || st.note || st.keypadDoor) {
   921→      if (st.prompt) st.setPrompt(null);
   922→      this.holdInteract = null;
   923→      return;
   924→    }
   925→    const p = this.player;
   926→    const look = p.lookDir;
   927→    let best: Interactable | null = null;
   928→    let bestScore = -1;
   929→    for (const inter of this.interactables) {
   930→      if (!inter.enabled()) continue;
   931→      const dx = inter.pos.x - p.pos.x;
   932→      const dz = inter.pos.z - p.pos.z;
   933→      const dy = (inter.pos.y ?? 1) - (p.pos.y + 1.5);
   934→      const d = Math.hypot(dx, dz, dy * 0.5);
   935→      if (d > inter.radius) continue;
   936→      let score = 1 - d / inter.radius;
   937→      if (inter.requireLook !== false) {
   938→        const len = Math.hypot(dx, dz) || 1;
   939→        const dot = (dx / len) * look.x + (dz / len) * look.z;
   940→        if (dot < 0.35) continue;
   941→        score += dot * 0.5;
   942→      }
   943→      if (score > bestScore) {
   944→        bestScore = score;
   945→        best = inter;
   946→      }
   947→    }
   948→    // recogidas automáticas (objetos de misión)
   949→    const item = this.missions.itemAt(p.pos, 0.95);
   950→    if (item) {
   951→      this.pickupItem(item.id, item.kind);
   952→    }
   953→    if (best) {
   954→      const holding = this.player.keys['KeyE'];
   955→      if (best.holdTime) {
   956→        if (holding) {
   957→          if (!this.holdInteract || this.holdInteract.inter !== best) {
   958→            this.holdInteract = { inter: best, progress: 0 };
   959→          }
   960→          this.holdInteract.progress += dt / best.holdTime;
   961→          this.addNoise(p.pos.x, p.pos.z, 13);
   962→          if (Math.random() < dt * 6) this.audio.repairClick();
   963→          if (this.holdInteract.progress >= 1) {
   964→            best.action();
   965→            this.holdInteract = null;
   966→            st.setPrompt(null);
   967→          } else {

  1325→  private render = () => {
  1326→    if (this.disposed) return;
  1327→    this.raf = requestAnimationFrame(this.render);
  1328→    const now = performance.now();
  1329→    let dt = (now - this.lastFrame) / 1000;
  1330→    this.lastFrame = now;
  1331→    if (dt > 0.06) dt = 0.06;
  1332→    if (dt <= 0) return;
  1333→
  1334→    // auto-ajuste de calidad
  1335→    this.fpsAcc += dt;
  1336→    this.fpsCount++;
  1337→    if (this.fpsAcc > 1) {
  1338→      const fps = this.fpsCount / this.fpsAcc;
  1339→      this.fpsAcc = 0;
  1340→      this.fpsCount = 0;
  1341→      if (this.autoQuality && this.mode === 'playing') {
  1342→        if (fps < 32) this.fpsLowTime += 1;
  1343→        else this.fpsLowTime = Math.max(0, this.fpsLowTime - 0.5);
  1344→        if (this.fpsLowTime >= 5) {
  1345→          if (this.quality === 'high') this.quality = 'medium';
  1346→          else if (this.quality === 'medium') this.quality = 'low';
  1347→          this.applyQuality();
  1348→          this.fpsLowTime = 0;
  1349→        }
  1350→      }
  1351→    }
  1352→
  1353→    if (this.mode === 'playing') {
  1354→      this.tick(dt);
  1355→    } else if (this.mode === 'dying') {
  1356→      this.updateDying(dt);
  1357→      this.world.update(dt, this.player.pos);
  1358→    } else if (this.mode === 'cinematic') {
  1359→      this.cine.update(dt);
  1360→      this.world.update(dt, this.camera.position);
  1361→    } else if (this.mode === 'menu' || this.mode === 'boot') {
  1362→      this.menuCamera(dt);
  1363→    } else if (this.mode === 'dead' || this.mode === 'complete' || this.mode === 'ending') {
  1364→      // escena congelada con la última cámara
  1365→      this.world.update(dt, this.camera.position);
  1366→    }
  1367→
  1368→    if (this.audio.available) {
  1369→      this.audio.update(dt);
  1370→      if (this.mode !== 'cinematic') {
  1371→        this.audio.updateListener(this.camera.position, this.camera.quaternion);
  1372→      } else {
  1373→        this.audio.updateListener(this.camera.position, this.camera.quaternion);
  1374→      }
  1375→    }
  1376→
  1377→    const st = useHud.getState();
  1378→    const tabletRendering = st.tabletOpen && this.mode === 'playing';
  1379→    // la tableta renderiza con más exposición (visión nocturna de las cámaras)
  1380→    this.renderer.toneMappingExposure = tabletRendering ? 3.0 : 1.0;
  1381→    // brazos del guardia en primera persona

  1383→    const activeCam = tabletRendering ? this.cams.camera : this.camera;
  1384→    this.renderer.render(this.scene, activeCam);
  1385→  };

  1387→  private tick(dt: number) {
  1388→    if (!this.chapterDef || !this.missions) return;
  1389→    const st = useHud.getState();
  1390→    this.time += dt;
  1391→
  1392→    // jugador (bloqueado si tablet/nota/teclado abiertos)
  1393→    const uiBlocked = st.tabletOpen || !!st.note || !!st.keypadDoor;
  1394→    if (!uiBlocked) {
  1395→      this.player.update(dt);
  1396→    } else {
  1397→      this.player.keys = {};
  1398→    }
  1399→
  1400→    // IA host actualizado
  1401→    const host = this.animatronics[0]?.host;
  1402→    if (host) {
  1403→      host.time = this.time;
  1404→      host.chapter = this.chapter;
  1405→      host.chapterTime = this.time;
  1406→      host.aggression = this.chapterDef.aggression * (this.trials.active ? 1.25 : 1);
  1407→    }
  1408→
  1409→    // animatrónicos (solo los del capítulo despiertan; el resto sigue dormido)
  1410→    const playerLike = this.player as unknown as PlayerLike;
  1411→    for (const a of this.animatronics) {
  1412→      if (!a.def.activeChapters.includes(this.chapter) && !a.active) {
  1413→        a.update(dt * 0.2, playerLike); // dormido: animación mínima
  1414→        continue;
  1415→      }
  1416→      a.update(dt, playerLike);
  1417→    }
  1418→
  1419→    // eventos
  1420→    this.events.update(dt);
  1421→    this.events.checkKitchenNoise(this.time / 60);
  1422→    if (!this.events.callDone && this.chapterDef.radio && this.time / 60 >= this.chapterDef.radio.atMinute) {
  1423→      this.events.triggerCall(this.chapterDef.radio.lines);
  1424→    }
  1425→    this.updateCall(dt);
  1426→
  1427→    // pruebas (trials): activación y efectos
  1428→    const room = this.world.roomIdAt(this.player.pos.x, this.player.pos.z);
  1429→    this.trials.update(dt, room);
  1430→
  1431→    // objetivos de desplazamiento
  1432→    const othersDoneFor = (excludeId: string) =>
  1433→      this.missions.objectives
  1434→        .filter((o) => o.def.id !== excludeId && !o.def.optional && o.def.type !== 'survive')
  1435→        .every((o) => o.done);
  1436→    const gotoChecks: Array<[string, string, boolean]> = [
  1437→      ['c1_atrium', 'atrium', false],
  1438→      ['c3_cove', 'pirateCove', true],
  1439→    ];
  1440→    for (const [objId, roomId, requireOthers] of gotoChecks) {
  1441→      const o = this.missions.objectives.find((x) => x.def.id === objId);
  1442→      if (o && !o.done && room === roomId && (!requireOthers || othersDoneFor(o.def.id))) {
  1443→        this.missions.completeObjective(objId);
  1444→        // susto guionizado al llegar a la cala
  1445→        if (objId === 'c3_cove') {
  1446→          const foxy = this.animatronics.find((a) => a.def.id === 'foxy');
  1447→          if (foxy && !foxy.active) {
  1448→            this.audio.stinger();
  1449→            this.player.setShake(0.6);
  1450→            this.notify('ALGO SE MUEVE TRAS EL TELÓN DE LA CALA...');
  1451→          }
  1452→        }
  1453→      }
  1454→    }
  1455→
  1456→    // energía
  1457→    let drain = 0.045;
  1458→    if (st.tabletOpen) drain += 0.29;
  1459→    if (this.player.flashlightOn) drain += 0.05;
  1460→    if (this.player.running) drain += 0.06;

  1462→    this.world.setPowerFactor(this.power > 25 ? 1 : this.power / 25);
  1463→    if (this.power <= 0) {
  1464→      // apagón total: solo quedan las rojas de emergencia

  1466→      this.powerOut = true;
  1467→    } else if (this.powerOut && this.power > 30) {
  1468→      // red recuperada: vuelve la tenue luz de la oficina
  1469→      this.powerOut = false;
  1470→      const office = this.world.rooms.get('office');
  1471→      if (office) office.lightOn = true;
  1472→    }
  1473→
  1474→    // puertas animadas
  1475→    for (const d of this.world.doors.values()) {
  1476→      if (d.angle !== d.targetAngle) {
  1477→        const diff = d.targetAngle - d.group.rotation.y;
  1478→        d.group.rotation.y += Math.max(-3.2 * dt, Math.min(3.2 * dt, diff));
  1479→        d.angle = d.group.rotation.y;
  1480→        d.collider.active = Math.abs(d.angle - (d.def.axis === 'x' ? 0 : Math.PI / 2)) < 0.5 && !d.open;
  1481→        if (Math.abs(diff) < 0.02) {
  1482→          d.angle = d.targetAngle;
  1483→          d.group.rotation.y = d.targetAngle;
  1484→        }
  1485→      }
  1486→    }
  1487→
  1488→    // tapas de conducto: Bonnie las tumba
  1489→    for (const v of this.world.vents.values()) {
  1490→      if (!v.coverClosed) continue;
  1491→      const bonnie = this.animatronics.find((a) => a.def.id === 'bonnie');
  1492→      if (bonnie && bonnie.active) {
  1493→        const px = v.def.coverAt === 'a' ? v.def.a[0] : v.def.b[0];
  1494→        const pz = v.def.coverAt === 'a' ? v.def.a[1] : v.def.b[1];
  1495→        const d = Math.hypot(bonnie.pos.x - px, bonnie.pos.z - pz);
  1496→        if (d < 1.6 && (bonnie.state === 'chase' || bonnie.state === 'hunt')) {

  1500→            v.coverTimer = 2.4;

  1505→            if (v.coverMesh) v.coverMesh.visible = false;
  1506→            if (v.coverCollider) v.coverCollider.active = false;
  1507→            this.nav.setVentBlocked(v.def.id, false);
  1508→            this.audio.doorSlam({ x: px, y: 0.6, z: pz });
  1509→            this.notify('¡TAPA DE CONDUCTO DERRIBADA!');
  1510→          }
  1511→        }
  1512→      }
  1513→    }
  1514→
  1515→    // detección de escondites
  1516→    if (this.player.hidden) {
  1517→      for (const a of this.animatronics) {
  1518→        if (!a.active || a.state === 'dormant') continue;
  1519→        const d = a.pos.distanceTo(this.player.pos);
  1520→        if (d < 2.0 && (a.state === 'search' || a.state === 'chase' || a.state === 'hunt')) {
  1521→          if (Math.random() < dt * 0.35) {
  1522→            const spot = mapData.hidingSpots.find((s) => s.id === this.player.hiddenSpotId);
  1523→            this.player.unhide(new THREE.Vector3(spot ? spot.pos[0] : this.player.pos.x, 0, spot ? spot.pos[1] : this.player.pos.z));
  1524→            a.state = 'chase';
  1525→            a.stateTimer = 0;
  1526→            this.notify('¡TE HAN ENCONTRADO!');
  1527→            break;
  1528→          }
  1529→        }
  1530→      }
  1531→    }
  1532→
  1533→    // peligro
  1534→    let danger = 0;
  1535→    let chase = false;
  1536→    for (const a of this.animatronics) {
  1537→      if (!a.active) continue;
  1538→      if (a.state === 'chase' || a.state === 'hunt' || a.state === 'attack') {
  1539→        chase = true;
  1540→        danger = Math.max(danger, 1);
  1541→      } else {
  1542→        const d = a.pos.distanceTo(this.player.pos);
  1543→        danger = Math.max(danger, Math.max(0, 1 - d / 14) * 0.5);

  1546→    this.danger = danger;
  1547→    this.audio.danger = danger;
  1548→    this.audio.setChase(chase);
  1549→
  1550→    // mundo (luces); en modo tableta se impulsa la sala del feed
  1551→    this.world.camView = st.tabletOpen;
  1552→    const focus = st.tabletOpen ? this.cams.camera.position : this.player.pos;
  1553→    this.world.update(dt, focus);
  1554→    this.updateDust(dt);
  1555→    this.updateInteractions(dt);
  1556→    this.cams.update(dt, this.animatronics, st.tabletOpen);
  1557→
  1558→    // fin de capítulo: todos los objetivos requeridos

  1563→
  1564→    // HUD a 10 Hz

  1568→      useHud.getState().syncHud({
  1569→        battery: this.player.flashlightBattery,
  1570→        stamina: this.player.stamina,
  1571→        power: this.power,
  1572→        maskOn: this.player.masked,
  1573→        flashlightOn: this.player.flashlightOn,
  1574→        crouching: this.player.crouching,
  1575→        inVent: this.player.inVent,
  1576→        danger,
  1577→        tabletGlitch: this.cams.glitchLevel > 0.55,
  1578→      });
  1579→    }
  1580→  }
  1581→
  1582→  /* ============ LIMPIEZA ============ */
  1583→
